private async sendDailyMarketReport(msk: Date) {
  try {
    const now = new Date();
    const dayStart = new Date(now.getTime() - 24 * 60 * 60 * 1000); // Последние 24 часа

    const dateStr = `${msk.getUTCDate().toString().padStart(2, "0")}.${(msk.getUTCMonth() + 1).toString().padStart(2, "0")}.${msk.getUTCFullYear()}`;

    // --- Сбор данных ---
    const currentPrice = aggregator.getPrimaryPrice();
    const fundingRate = aggregator.getFundingRate();
    const oiData = coinglassService.getCachedOpenInterest();
    const liqData = coinglassService.getCachedHeatmap();
    const longShortRatio = coinglassService.getCachedLongShortRatio();
    const cvdData = cvdService.getCachedCvdData();

    // Расчет изменения OI за 24 часа
    let oiChange24h = 0;
    if (oiData && oiData.history && oiData.history.length > 0) {
      const currentOi = oiData.history[oiData.history.length - 1].oi;
      const oi24hAgoEntry = oiData.history.find(entry => (now.getTime() - entry.timestamp) <= 24 * 60 * 60 * 1000);
      if (oi24hAgoEntry) {
        oiChange24h = (currentOi - oi24hAgoEntry.oi) / oi24hAgoEntry.oi * 100;
      }
    }

    // Расчет общих ликвидаций за 24 часа
    let totalLiquidations24h = 0;
    if (liqData && liqData.history) {
      const liq24hAgoEntry = liqData.history.find(entry => (now.getTime() - entry.timestamp) <= 24 * 60 * 60 * 1000);
      if (liq24hAgoEntry) {
        // Предполагаем, что history содержит суммарные ликвидации за период, или нужно агрегировать
        // Для простоты, возьмем разницу между текущим и 24ч назад, если это кумулятивные данные
        // Если это данные по часам, то нужно суммировать за 24 часа
        // Здесь требуется уточнение структуры liqData.history
        // Для примера, возьмем последние 24 записи и просуммируем их, если это часовые данные
        const relevantLiqs = liqData.history.filter(entry => (now.getTime() - entry.timestamp) <= 24 * 60 * 60 * 1000);
        totalLiquidations24h = relevantLiqs.reduce((sum, entry) => sum + entry.longLiqUsd + entry.shortLiqUsd, 0);
      }
    }

    // Определение сентимента по Long/Short Ratio
    let sentiment = "Нейтральный";
    let longShortPct = 0;
    if (longShortRatio) {
      longShortPct = longShortRatio.longShortRatio * 100;
      if (longShortRatio.longShortRatio > 1.05) sentiment = "Бычий (лонги преобладают)";
      else if (longShortRatio.longShortRatio < 0.95) sentiment = "Медвежий (шорты преобладают)";
    }

    // Определение сентимента по CVD (простое)
    let cvdSentiment = "Нейтральный";
    if (cvdData && cvdData.alignment) {
      if (cvdData.alignment.score > 0.2) cvdSentiment = "Бычий (покупатели активны)";
      else if (cvdData.alignment.score < -0.2) cvdSentiment = "Медвежий (продавцы активны)";
    }

    const msg =
      `☀️ <b>ЕЖЕДНЕВНЫЙ РЫНОЧНЫЙ ОТЧЕТ — ${dateStr} (11:00 МСК)</b>\n` +
      `━━━━━━━━━━━━━━━━━━━\n` +
      `💰 Текущая цена BTC: <b>$${currentPrice.toFixed(1)}</b>\n` +
      `📊 Сентимент (Long/Short Ratio): <b>${longShortPct.toFixed(2)}%</b> (${sentiment})\n` +
      `📈 Сентимент (CVD): <b>${cvdSentiment}</b>\n` +
      `💸 Ставка фандинга: <b>${(fundingRate * 100).toFixed(4)}%</b>\n` +
      `🔄 Изменение OI (24ч): <b>${oiChange24h >= 0 ? "+" : ""}${oiChange24h.toFixed(2)}%</b>\n` +
      `💥 Ликвидации (24ч): <b>$${(totalLiquidations24h / 1_000_000).toFixed(2)}M</b>\n` +
      `━━━━━━━━━━━━━━━━━━━\n` +
      `#BTC #рынок #отчет #daily`
    ;

    telegramService.notifyChannel(msg);
    logger.info({ date: dateStr }, "Daily market report sent");
  } catch (e) {
    logger.error({ err: e }, "Failed to send daily market report");
  }
}



•
Объяснение: Эта функция собирает различные рыночные данные, форматирует их в читабельный отчет и отправляет в Telegram-канал. Обратите внимание, что для totalLiquidations24h я сделал предположение о структуре liqData.history. Если она другая, возможно, потребуется небольшая корректировка этой части кода.
