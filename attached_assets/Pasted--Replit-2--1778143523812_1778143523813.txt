3.
Измените логику fetch1DCandles() для получения данных с 2013 года.

Пояснение: Поскольку CryptoCompare имеет ограничение limit=2000, для получения данных с 2013 года потребуется несколько запросов. Мы будем делать запросы, начиная с текущей даты и двигаясь назад во времени, пока не достигнем 2013 года или не соберем достаточно данных.

Замените всю функцию fetch1DCandles() на следующую:

TypeScript


private async fetch1DCandles(): Promise<OhlcvCandle[]> {
    const allCandles: OhlcvCandle[] = [];
    let hasMore = true;
    let toTs = Math.floor(Date.now() / 1000); // Current timestamp in seconds
    const targetYear = 2013; // Target year for historical data

    logger.info("Fetching 1D candles from CryptoCompare (multiple requests for deep history)...");

    while (hasMore) {
        try {
            const ccRes = await axios.get(`https://min-api.cryptocompare.com/data/histoday?fsym=BTC&tsym=USDT&limit=2000&toTs=${toTs}&e=Binance`, {
                headers: { 'authorization': `Apikey ${process.env.CRYPTOCOMPARE_API_KEY}` }
            } );

            if (ccRes.data && ccRes.data.Data && ccRes.data.Data.length > 0) {
                const newCandles = ccRes.data.Data.map((c: any) => ({
                    time: c.time * 1000, // Convert to milliseconds
                    open: c.open,
                    high: c.high,
                    low: c.low,
                    close: c.close,
                    volume: c.volumeto,
                })).filter((c: OhlcvCandle) => new Date(c.time).getFullYear() >= targetYear);

                allCandles.unshift(...newCandles); // Add to the beginning to maintain chronological order

                // Find the timestamp of the oldest candle fetched in this batch
                const oldestCandleTime = newCandles.length > 0 ? newCandles[0].time : 0;
                toTs = Math.floor(oldestCandleTime / 1000) - 1; // Request data before this timestamp

                // Stop if we reached the target year or no more data
                if (new Date(oldestCandleTime).getFullYear() <= targetYear || ccRes.data.Data.length < 2000) {
                    hasMore = false;
                }
                logger.debug(`Fetched ${newCandles.length} 1D candles. Total: ${allCandles.length}. Oldest: ${new Date(allCandles[0].time).toISOString()}`);

            } else {
                hasMore = false; // No more data from CryptoCompare
            }
        } catch (e) {
            logger.error({ err: e }, "Failed to fetch 1D candles from CryptoCompare");
            hasMore = false; // Stop on error
        }
    }

    if (allCandles.length === 0) {
        logger.warn("CryptoCompare returned no 1D candles. Falling back to aggregator.");
        // Fallback to Bybit if CryptoCompare fails completely
        const bybitCandles = await aggregator.fetchKlines("Bybit", "BTCUSDT", "1d", 500);
        if (bybitCandles.length > 0) {
            return bybitCandles.map(c => ({ ...c, volume: c.volume ?? 0 })); // Ensure volume is number
        }
    }

    // Ensure candles are sorted by time ascending and unique
    const uniqueCandles = Array.from(new Map(allCandles.map(c => [c.time, c])).values());
    uniqueCandles.sort((a, b) => a.time - b.time);

    logger.info(`Finished fetching 1D candles. Total: ${uniqueCandles.length}. From: ${new Date(uniqueCandles[0].time).toISOString()} To: ${new Date(uniqueCandles[uniqueCandles.length - 1].time).toISOString()}`);
    return uniqueCandles;
}
