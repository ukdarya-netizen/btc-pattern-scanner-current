2.3. Внедрение Мониторинга Стакана (Order Book Imbalance) (src/services/orderbookService.ts):

Цель: Добавить фильтр, который проверяет перекос лимитных заявок в стакане. Если есть "плиты" против направления сигнала, сигнал отклоняется.

Откройте файл src/services/orderbookService.ts (или создайте, если нет).

Добавьте новую функцию для проверки дисбаланса:

TypeScript


// src/services/orderbookService.ts

export interface OrderBookImbalanceResult {
  pass: boolean;
  reason: string;
}

export function checkImbalance(direction: 'long' | 'short'): OrderBookImbalanceResult {
  // Здесь должна быть логика получения данных стакана и их анализа.
  // Пример: получить данные с биржи, посчитать объем на 1% от текущей цены в обе стороны.
  // Для примера, заглушка:
  const bidVolume = getOrderBookVolume('bid', 0.005); // Объем покупок на 0.5% ниже цены
  const askVolume = getOrderBookVolume('ask', 0.005); // Объем продаж на 0.5% выше цены

  const imbalanceThreshold = 2.0; // Например, если объем одной стороны в 2 раза больше другой

  if (direction === 'long') {
    if (askVolume > bidVolume * imbalanceThreshold) {
      return { pass: false, reason: `Сильное сопротивление в стакане (${(askVolume / bidVolume).toFixed(1)}x Ask > Bid)` };
    }
  } else if (direction === 'short') {
    if (bidVolume > askVolume * imbalanceThreshold) {
      return { pass: false, reason: `Сильная поддержка в стакане (${(bidVolume / askVolume).toFixed(1)}x Bid > Ask)` };
    }
  }

  return { pass: true, reason: "" };
}
