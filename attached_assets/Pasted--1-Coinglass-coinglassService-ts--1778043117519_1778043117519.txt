  price: number,
  buckets: LiqHeatmapBucket[]  // реальные данные из Coinglass

  // LONG: ищем кластер ЛОНГОВ ниже цены
  // Это зона где лонгистов вынесет каскадом — выходим ДО неё

  // При нескольких кандидатах — брать ближайший к цене значимый кластер
  // (не самый крупный — а ближайший, потому что он сработает первым)

ИЗМЕНЕНИЕ 10: Cross-Exchange Divergence
Файл: signal.ts
Добавить display-only компонент:
typescriptfunction calcCrossExchangeImbalance(): SignalComponent {
  const agg = aggregator.getAggregatedOrderbook();
  if (!agg || Object.keys(agg.byExchange).length < 2) {
    return { name: "Cross-Exchange Divergence", score: 0, weight: 0, 
             description: "Нужно 2+ биржи" };
  }

  // Считаем imbalance отдельно для каждой биржи
  const imbalances: Record<string, number> = {};
  for (const [exchange, ob] of Object.entries(agg.byExchange)) {
    imbalances[exchange] = ob.imbalance;
  }

  const values = Object.values(imbalances);
  const maxDivergence = Math.max(...values) - Math.min(...values);
  
  // Если расхождение > 0.3 — манипуляция на одной из бирж
  // Среднее imbalance всё равно используем как score
  const avgImbalance = values.reduce((s, v) => s + v, 0) / values.length;
  
  const desc = Object.entries(imbalances)
    .map(([ex, im]) => `${ex}: ${(im*100).toFixed(1)}%`)
    .join(' | ');
  
  const warning = maxDivergence > 0.3 
    ? ` ⚠️ Расхождение ${(maxDivergence*100).toFixed(0)}% — возможна манипуляция` 
    : '';

  return {
    name: "Cross-Exchange Divergence",
    score: avgImbalance,
    weight: 0,  // display-only
    description: desc + warning,
  };
}
