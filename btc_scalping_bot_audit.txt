import { aggregator } from "./exchanges/aggregator";

export interface SignalComponent {
  name: string;
  score: number;
  weight: number;
  description: string;
}

export interface GateResult {
  name: string;
  passed: boolean;
  reason: string;
}

export interface SignalResult {
  score: number;
  direction: "long" | "short" | "neutral";
  confidence: number;
  components: SignalComponent[];
  gates: GateResult[];
  allGatesPassed: boolean;
  suggestedLeverage: number;
  suggestedSl: number;
  suggestedTp: number;
  atrPct: number;
  vwap: number;
  updatedAt: string;
  tpSource?: string;
  slSource?: string;
}

// Weights: Pattern 0.30 + CVD 0.20 + OB 0.15 + Momentum 0.13 + VWAP 0.10 + EMA 0.07 + Liquidations 0.05 = 1.00
// Funding + OI: weight 0 (display only)

function clamp(val: number, min = -1, max = 1): number {
  return Math.max(min, Math.min(max, val));
}

function calcPatternSignal(): SignalComponent {
  if (!patternService.isReady()) {
    return { name: "Historical Pattern", score: 0, weight: 0.30, description: "Загрузка исторических данных..." };
  }

  const result = patternService.getResult()!;
  const score = parseFloat(result.signal.toFixed(4));
  const bull = result.bullishMatches;
  const bear = result.bearishMatches;
  const neutral = result.topMatches.length - bull - bear;
  const topMatch = result.topMatches[0];
  const simPct = topMatch ? (topMatch.similarity * 100).toFixed(1) : "0";
  const fwdLabel = topMatch
    ? `${topMatch.forwardPct >= 0 ? "+" : ""}${topMatch.forwardPct.toFixed(2)}% за 3д`
    : "";

  return {
    name: "Historical Pattern",
    score: clamp(score),
    weight: 0.30,
    description: `Топ совпадение: ${topMatch?.startDate ?? "—"} (r=${simPct}%) → ${fwdLabel} | Бычьих: ${bull} / Медвежьих: ${bear} / Нейтр: ${neutral} из ${result.topMatches.length}`,
  };
}

function calcCvdSignal(): SignalComponent {
  const points = cvdService.getPoints();
  const trend = cvdService.getTrend();
  let score = 0;
  if (trend === "bullish") score = 0.6;
  else if (trend === "bearish") score = -0.6;

  if (points.length >= 5) {
    const recent = points.slice(-5);
    const avg = recent.reduce((s, p) => s + p.delta, 0) / recent.length;
    const magnitude = clamp(avg / 5, -0.4, 0.4);
    score = clamp(score + magnitude);
  }

  return {
    name: "CVD Trend",
    score,
    weight: 0.20,
    description: `Trend: ${trend}, CVD: ${cvdService.getCurrentCvd().toFixed(2)}`,
  };
}

function calcOrderbookImbalance(): SignalComponent {
  const ob = aggregator.getAggregatedOrderbook();
  if (!ob) return { name: "Order Book Imbalance", score: 0, weight: 0.15, description: "No data" };

  const bidVol = ob.mergedBids.slice(0, 10).reduce((s, l) => s + l.size, 0);
  const askVol = ob.mergedAsks.slice(0, 10).reduce((s, l) => s + l.size, 0);
  const total = bidVol + askVol;
  const score = total > 0 ? clamp((bidVol - askVol) / total * 3) : 0;

  const exchanges = Object.keys(ob.byExchange).join(", ");
  return {
    name: "Order Book Imbalance",
    score,
    weight: 0.15,
    description: `Bid: ${bidVol.toFixed(2)} BTC vs Ask: ${askVol.toFixed(2)} BTC (${exchanges})`,
  };
}

function calcMomentumSignal(): SignalComponent {
  const trades = aggregator.getAllTrades().slice(0, 60);
  if (trades.length === 0) return { name: "Trade Momentum", score: 0, weight: 0.13, description: "No data" };

  const exchanges = [...new Set(trades.map(t => t.exchange))].join("+");
  const buyVol = trades.filter(t => t.side === "buy").reduce((s, t) => s + t.size, 0);
  const sellVol = trades.filter(t => t.side === "sell").reduce((s, t) => s + t.size, 0);
  const total = buyVol + sellVol;
  const score = total > 0 ? clamp((buyVol - sellVol) / total * 2) : 0;

  return {
    name: "Trade Momentum",
    score,
    weight: 0.13,
    description: `Buy: ${buyVol.toFixed(2)} BTC vs Sell: ${sellVol.toFixed(2)} BTC (${exchanges}, last 60 trades)`,
  };
}

function calcVwapSignal(): SignalComponent {
  const price = aggregator.getPrimaryPrice();
  const vwap = candleService.getVwap();
  const atr = candleService.getAtr();

  if (vwap === 0 || price === 0) {
    return { name: "VWAP Deviation", score: 0, weight: 0.10, description: "Accumulating VWAP data..." };
  }

  const deviation = (price - vwap) / vwap;
  const score = clamp(deviation / 0.004 * 0.8);
  const sigma = atr > 0 ? Math.abs(price - vwap) / atr : 0;
  const direction = deviation >= 0 ? "above" : "below";
  const urgency = sigma > 2 ? " — extreme, mean reversion likely" : sigma > 1 ? " — extended" : "";

  return {
    name: "VWAP Deviation",
    score,
    weight: 0.10,
    description: `$${price.toFixed(1)} ${direction} VWAP $${vwap.toFixed(1)} by ${(Math.abs(deviation) * 100).toFixed(3)}% (${sigma.toFixed(1)}σ)${urgency}`,
  };
}

function calcEmaSignal(): SignalComponent {
  const ema9 = candleService.getEma(9);
  const ema21 = candleService.getEma(21);
  const price = aggregator.getPrimaryPrice();
  const candleCount = candleService.getCandleCount();

  if (ema9 === 0 || ema21 === 0 || candleCount < 5) {
    return { name: "EMA 9/21", score: 0, weight: 0.07, description: `Building candle history... (${candleCount} candles)` };
  }

  let score = 0;
  if (ema9 > ema21) score = 0.5;
  else if (ema9 < ema21) score = -0.5;
  if (price > ema9 && score > 0) score = clamp(score + 0.3);
  if (price < ema9 && score < 0) score = clamp(score - 0.3);

  const crossLabel = ema9 > ema21 ? "Bullish (9>21)" : "Bearish (9<21)";
  const priceVsEma = price > ema9 ? "Price above EMA9" : "Price below EMA9";

  return {
    name: "EMA 9/21",
    score: clamp(score),
    weight: 0.07,
    description: `${crossLabel} | ${priceVsEma} | EMA9: $${ema9.toFixed(1)} EMA21: $${ema21.toFixed(1)}`,
  };
}

function calcLiquidationSignal(): SignalComponent {
  const candles = coinalyzeService.getLatest2hCandles();

  if (candles.length < 6) {
    return {
      name: "Ликвидации (Coinalyze)",
      score: 0,
      weight: 0.05,
      description: coinalyzeService.isReady() ? "Недостаточно данных" : "Загрузка...",
    };
  }

  // Recent = last 2 candles (10 min), baseline = previous 4-12 candles
  const recent = candles.slice(-2);
  const baseline = candles.slice(-12, -2);

  const recentLong = recent.reduce((s, c) => s + c.longBtc, 0);
  const recentShort = recent.reduce((s, c) => s + c.shortBtc, 0);
  const recentTotal = recentLong + recentShort;

  const baseLong = baseline.reduce((s, c) => s + c.longBtc, 0) / Math.max(baseline.length, 1);
  const baseShort = baseline.reduce((s, c) => s + c.shortBtc, 0) / Math.max(baseline.length, 1);

  // Per-candle averages for recent period
  const recentLongAvg = recentLong / recent.length;
  const recentShortAvg = recentShort / recent.length;

  // Spike ratios vs baseline (capped to avoid division by zero)
  const longSpike = baseLong > 0.001 ? recentLongAvg / baseLong : recentLongAvg > 0.01 ? 2 : 1;
  const shortSpike = baseShort > 0.001 ? recentShortAvg / baseShort : recentShortAvg > 0.01 ? 2 : 1;

  // Net ratio: positive = short spike (bullish), negative = long spike (bearish)
  // Short liquidations → price went up (shorts squeezed) → bullish signal
  // Long liquidations  → price went down (longs stopped out) → bearish signal
  let score = 0;
  const spikeThreshold = 1.5; // need 50% above baseline to count

  if (shortSpike >= spikeThreshold || longSpike >= spikeThreshold) {
    const netSpike = shortSpike - longSpike;
    score = clamp(netSpike / 3); // normalize: spike ratio diff of 3x → full score
  }

  // Also factor in absolute recent ratio
  const absRatio = recentTotal > 0.001
    ? (recentShort - recentLong) / recentTotal
    : 0;
  score = clamp(score * 0.6 + absRatio * 0.4);

  const shortSpikeLbl = shortSpike >= spikeThreshold ? ` ⚡x${shortSpike.toFixed(1)}` : "";
  const longSpikeLbl = longSpike >= spikeThreshold ? ` ⚡x${longSpike.toFixed(1)}` : "";
  const desc = `Лонги: ${recentLong.toFixed(2)} BTC${longSpikeLbl} | Шорты: ${recentShort.toFixed(2)} BTC${shortSpikeLbl} (последние 10 мин, все биржи)`;

  return {
    name: "Ликвидации (Coinalyze)",
    score,
    weight: 0.05,
    description: desc,
  };
}

// Display-only (weight 0) — shown on dashboard but don't affect score
function calcFundingDisplay(): SignalComponent {
  const fundingRate = aggregator.getPrimaryFundingRate();
  const exchanges = aggregator.getActiveExchangeNames().join("+");
  let score = 0;
  if (fundingRate > 0.001) score = -0.5;
  else if (fundingRate > 0.0005) score = -0.25;
  else if (fundingRate < -0.001) score = 0.5;
  else if (fundingRate < -0.0005) score = 0.25;

  return {
    name: "Funding Rate",
    score,
    weight: 0,
    description: `Rate: ${(fundingRate * 100).toFixed(4)}% avg (${exchanges}) — ${score !== 0 ? (score > 0 ? "аномально низкий (лонги платят мало)" : "аномально высокий (шорты платят)") : "норма"}`,
  };
}

function calcOpenInterestDisplay(): SignalComponent {
  const oi = aggregator.getPrimaryOpenInterest();
  const price = aggregator.getPrimaryPrice();
  const exchanges = aggregator.getActiveExchangeNames().join("+");
  if (price === 0 || oi === 0) return { name: "Open Interest", score: 0, weight: 0, description: "Нет данных OI" };

  const oiUsd = oi * price;
  const cvdTrend = cvdService.getTrend();
  return {
    name: "Open Interest",
    score: 0,
    weight: 0,
    description: `OI: $${(oiUsd / 1e6).toFixed(1)}M (${exchanges}) — тренд CVD: ${cvdTrend}`,
  };
}

function calcCrossExchangeDisplay(): SignalComponent {
  const agg = aggregator.getAggregatedTicker();
  if (!agg || Object.keys(agg.prices).length < 2) {
    return { name: "Cross-Exchange Spread", score: 0, weight: 0, description: "Need 2+ exchanges" };

  const priceParts = Object.entries(agg.prices)

    .join(" | ");
  return { name: "Cross-Exchange Spread", score: 0, weight: 0, description: priceParts };
}

class SignalEngine {
  compute(): SignalResult {
    // Weighted components (affect score)
    const weighted: SignalComponent[] = [
      calcPatternSignal(),       // 30% — PRIMARY
      calcCvdSignal(),           // 20%
      calcOrderbookImbalance(),  // 15%
      calcMomentumSignal(),      // 13%
      calcVwapSignal(),          // 10%
      calcEmaSignal(),           //  7%
      calcLiquidationSignal(),   //  5%
    ];

    // Display-only (weight 0, shown for context)
    const displayOnly: SignalComponent[] = [
      calcFundingDisplay(),
      calcOpenInterestDisplay(),
      calcCrossExchangeDisplay(),
    ];

    const components = [...weighted, ...displayOnly];

    const totalWeight = weighted.reduce((s, c) => s + c.weight, 0);
    const score = clamp(
      totalWeight > 0
        ? weighted.reduce((s, c) => s + c.score * c.weight, 0) / totalWeight
        : 0
    );

    const confidence = clamp(Math.abs(score) * 1.5, 0, 1);

    // Pattern is PRIMARY — if it's ready and strong, its direction overrides composite
    const MIN_PATTERN_SIGNAL = 0.20;
    const MIN_PATTERN_SIMILARITY = 0.75;
    const patternCompEarly = weighted.find(c => c.name === "Historical Pattern");
    const patternResultEarly = patternService.isReady() ? patternService.getResult() : null;
    const patternTopSimEarly = patternResultEarly?.topMatches[0]?.similarity ?? 0;
    const patternScoreEarly = patternCompEarly?.score ?? 0;
    const patternStrongEnough =
      patternService.isReady() &&
      Math.abs(patternScoreEarly) >= MIN_PATTERN_SIGNAL &&
      patternTopSimEarly >= MIN_PATTERN_SIMILARITY;

    let direction: "long" | "short" | "neutral" = "neutral";
    if (patternStrongEnough) {
      // Pattern determines direction
      direction = patternScoreEarly > 0 ? "long" : "short";
    } else if (score > 0.25) {
      direction = "long";
    } else if (score < -0.25) {
      direction = "short";
    }

    const price = aggregator.getPrimaryPrice() || 100000;
    const atr = candleService.getAtr();
    const atrPct = atr > 0 ? (atr / price) * 100 : 0;
    const vwap = candleService.getVwap();

    let leverage: number;
    if (atrPct > 0.4) leverage = 75;
    else if (atrPct > 0.2) leverage = 85;
    else leverage = 100;

    const atrSlDistance = atr > 0 ? Math.max(atr * 1.5, price * 0.0012) : price * 0.0015;

    // ── Cluster-based TP / SL using liquidation map ──
    const MIN_TP_CLUSTER_USD = 3_000_000;  // need $3M+ cluster to use as TP target
    const MIN_SL_CLUSTER_USD = 1_500_000;  // need $1.5M+ cluster as SL danger zone
    const MAX_TP_RANGE = 0.08;             // look up to 8% away for TP
    const MAX_SL_RANGE = 0.05;             // look up to 5% away for SL reference

    const liqMap = coinalyzeService.getCachedMap();
    const buckets = liqMap?.buckets ?? [];

    let suggestedTp: number;
    let suggestedSl: number;
    let tpSource: string;
    let slSource: string;

    if (direction === "long") {
      // TP: nearest significant SHORT liq cluster ABOVE price (shorts squeezed = upside magnet)
      const tpCluster = buckets
        .filter(b => b.price > price && b.price <= price * (1 + MAX_TP_RANGE) && b.shortLiqUsd >= MIN_TP_CLUSTER_USD)
        .sort((a, b) => a.price - b.price)[0];

      if (tpCluster) {
        suggestedTp = parseFloat((tpCluster.price * 0.999).toFixed(2)); // just before cluster
        tpSource = `Кластер шортов $${tpCluster.price.toFixed(0)} (${(tpCluster.shortLiqUsd / 1e6).toFixed(1)}M шортов)`;
      } else {
        suggestedTp = parseFloat((price + atrSlDistance * 1.5).toFixed(2));
        tpSource = "ATR × 1.5 (нет кластера в диапазоне)";
      }

      // SL: nearest large LONG liq cluster BELOW price (cascade danger zone)
      const slCluster = buckets
        .filter(b => b.price < price && b.price >= price * (1 - MAX_SL_RANGE) && b.longLiqUsd >= MIN_SL_CLUSTER_USD)
        .sort((a, b) => b.price - a.price)[0]; // closest to current price

      const atrSlPrice = price - atrSlDistance;
      if (slCluster && slCluster.price > atrSlPrice) {
        // Cluster is between price and ATR SL → exit BEFORE cascade (slightly above cluster)
        suggestedSl = parseFloat((slCluster.price * 1.0005).toFixed(2));
        slSource = `Выше кластера лонгов $${slCluster.price.toFixed(0)} (${(slCluster.longLiqUsd / 1e6).toFixed(1)}M) — до каскада`;
      } else {
        suggestedSl = parseFloat(atrSlPrice.toFixed(2));
        slSource = "ATR × 1.5 (кластер далеко или отсутствует)";
      }

    } else if (direction === "short") {
      // TP: nearest significant LONG liq cluster BELOW price (longs stopped = downside magnet)
      const tpCluster = buckets
        .filter(b => b.price < price && b.price >= price * (1 - MAX_TP_RANGE) && b.longLiqUsd >= MIN_TP_CLUSTER_USD)
        .sort((a, b) => b.price - a.price)[0]; // closest below

      if (tpCluster) {
        suggestedTp = parseFloat((tpCluster.price * 1.001).toFixed(2)); // just above cluster
        tpSource = `Кластер лонгов $${tpCluster.price.toFixed(0)} (${(tpCluster.longLiqUsd / 1e6).toFixed(1)}M лонгов)`;
      } else {
        suggestedTp = parseFloat((price - atrSlDistance * 1.5).toFixed(2));
        tpSource = "ATR × 1.5 (нет кластера в диапазоне)";
      }

      // SL: nearest large SHORT liq cluster ABOVE price (cascade danger for shorts)
      const slCluster = buckets
        .filter(b => b.price > price && b.price <= price * (1 + MAX_SL_RANGE) && b.shortLiqUsd >= MIN_SL_CLUSTER_USD)
        .sort((a, b) => a.price - b.price)[0]; // closest above

      const atrSlPrice = price + atrSlDistance;
      if (slCluster && slCluster.price < atrSlPrice) {
        suggestedSl = parseFloat((slCluster.price * 0.9995).toFixed(2));
        slSource = `Ниже кластера шортов $${slCluster.price.toFixed(0)} (${(slCluster.shortLiqUsd / 1e6).toFixed(1)}M) — до каскада`;
      } else {
        suggestedSl = parseFloat(atrSlPrice.toFixed(2));
        slSource = "ATR × 1.5 (кластер далеко или отсутствует)";
      }

    } else {
      suggestedTp = parseFloat((price + atrSlDistance * 1.5).toFixed(2));
      suggestedSl = parseFloat((price - atrSlDistance).toFixed(2));
      tpSource = "ATR (нейтральное направление)";
      slSource = "ATR (нейтральное направление)";
    }

    // ── Entry gates ──
    const dirSign = direction === "long" ? 1 : direction === "short" ? -1 : 0;
    const cvdComp      = weighted.find(c => c.name === "CVD Trend");
    const momentumComp = weighted.find(c => c.name === "Trade Momentum");
    const patternComp  = weighted.find(c => c.name === "Historical Pattern");

    // Gate 1 (PRIMARY): Pattern — correlation ≥ 75%, signal strength ≥ 0.20, direction match
    let patternPassed = false;
    let patternReason = "";
    if (!patternService.isReady()) {
      patternReason = "Исторические данные загружаются";
    } else if (direction === "neutral") {
      patternReason = "Направление нейтральное";
    } else {
      const pr = patternService.getResult()!;
      const topSim = pr.topMatches[0]?.similarity ?? 0;
      const pScore = patternComp?.score ?? 0;
      const pAbs = Math.abs(pScore);
      const dirOk = pScore * dirSign > 0;
      const simOk = topSim >= MIN_PATTERN_SIMILARITY;
      const strengthOk = pAbs >= MIN_PATTERN_SIGNAL;
      patternPassed = dirOk && simOk && strengthOk;
      const simStr = `топ r=${(topSim * 100).toFixed(1)}% ${simOk ? "✓" : `< ${(MIN_PATTERN_SIMILARITY * 100).toFixed(0)}% ✗`}`;
      const strStr = `сила |${pScore.toFixed(3)}| ${strengthOk ? "✓" : `< ${MIN_PATTERN_SIGNAL} ✗`}`;
      const dirStr = dirOk ? "направление ✓" : "направление ✗";
      patternReason = `${simStr} | ${strStr} | ${dirStr}`;
    }
    const gate1: GateResult = { name: "Паттерн-матчинг (первичный)", passed: patternPassed, reason: patternReason };

    // Gate 2: Order Book Imbalance — стакан должен подтверждать направление
    const obComp = weighted.find(c => c.name === "Order Book Imbalance");
    let obPassed = false;
    let obReason = "";
    if (direction === "neutral") {
      obReason = "Направление нейтральное";
    } else {
      obPassed = obComp ? obComp.score * dirSign > 0 : false;
      const obStr = `Стакан ${obComp ? (obComp.score >= 0 ? "+" : "") + obComp.score.toFixed(2) : "?"} ${obPassed ? "✓" : "✗"}`;
      obReason = obStr + (obComp ? ` (bid/ask ${dirSign > 0 ? "перевес bid" : "перевес ask"})` : " — нет данных");
    }
    const gate2: GateResult = { name: "Стакан (Order Book)", passed: obPassed, reason: obReason };

    const gates: GateResult[] = [gate1, gate2];
    // Both gates required: Pattern (primary) + Order Book (secondary)
    const allGatesPassed = direction !== "neutral" && gate1.passed && gate2.passed;

    return {
      score: parseFloat(score.toFixed(4)),
      direction,
      confidence: parseFloat(confidence.toFixed(4)),
      components,
      gates,
      allGatesPassed,
      suggestedLeverage: leverage,
      suggestedSl,
      suggestedTp,
      atrPct: parseFloat(atrPct.toFixed(4)),
      vwap: parseFloat(vwap.toFixed(2)),
      updatedAt: new Date().toISOString(),
      tpSource,
      slSource,
    };
  }
}

class SignalTracker {
  private active: ActiveSignal | null = null;
  private lastSignalAt = 0;
  private tickInterval: ReturnType<typeof setInterval> | null = null;

  start() {
    if (this.tickInterval) return;
    this.tickInterval = setInterval(() => this.tick(), 5000);
    logger.info("SignalTracker started — tracking signal performance 24/7");
  }

  stop() {
    if (this.tickInterval) {
      clearInterval(this.tickInterval);
      this.tickInterval = null;
    }
  }

  getActive(): ActiveSignal | null {
    return this.active;
  }

  private tick() {
    const price = aggregator.getPrimaryPrice();
    if (price === 0) return;

    // Check if active signal hit TP or SL
    if (this.active) {
      const { side, entryPrice, stopLoss, takeProfit, leverage } = this.active;
      const hitTp = side === "long" ? price >= takeProfit : price <= takeProfit;
      const hitSl = side === "long" ? price <= stopLoss : price >= stopLoss;

      if (hitTp || hitSl) {
        const exitPrice = hitTp ? takeProfit : stopLoss;
        const reason = hitTp ? "tp" : "sl";
        void this.closeSignal(exitPrice, reason);
        return;
      }
    }

    // Check for new signal
    const now = Date.now();
    if (this.active) return;
    if (now - this.lastSignalAt < MIN_SIGNAL_INTERVAL_MS) return;

    const signal = signalEngine.compute();
    if (!signal.allGatesPassed) return;
    if (signal.direction === "neutral") return;

    this.active = {
      id: `sig-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
      side: signal.direction as "long" | "short",
      entryPrice: price,
      stopLoss: signal.suggestedSl,
      takeProfit: signal.suggestedTp,
      leverage: signal.suggestedLeverage,
      openedAt: new Date(),
    };
    this.lastSignalAt = now;

    const slPct = Math.abs((signal.suggestedSl - price) / price * 100).toFixed(2);
    const tpPct = Math.abs((signal.suggestedTp - price) / price * 100).toFixed(2);

    logger.info({
      side: signal.direction,
      price,
      sl: signal.suggestedSl,
      tp: signal.suggestedTp,
      leverage: signal.suggestedLeverage,
      score: signal.score,
    }, "SignalTracker: signal opened");

    telegramService.notifyChannel(
      `📡 <b>СИГНАЛ ${signal.direction === "long" ? "🟢 LONG" : "🔴 SHORT"}</b>\n` +
      `━━━━━━━━━━━━━━━━━━━\n` +
      `💰 Цена входа: <b>$${price.toFixed(1)}</b>\n` +
      `⚡ Плечо: <b>${signal.suggestedLeverage}x</b>\n` +
      `💪 Уверенность: <b>${(signal.confidence * 100).toFixed(0)}%</b>\n` +
      `🛑 SL: <b>$${signal.suggestedSl.toFixed(1)}</b> (-${slPct}%)\n` +
      `🎯 TP: <b>$${signal.suggestedTp.toFixed(1)}</b> (+${tpPct}%)\n` +
      `📊 Score: ${signal.score >= 0 ? "+" : ""}${signal.score.toFixed(3)}\n` +

      `#BTC #bitcoin #scalping`

  }

  private async closeSignal(exitPrice: number, reason: "tp" | "sl") {

    const { id, side, entryPrice, leverage, openedAt } = this.active;

    const dirMult = side === "long" ? 1 : -1;
    const pnlPct = (exitPrice - entryPrice) / entryPrice * dirMult * leverage;
    const notionalUsd = entryPrice * VIRTUAL_SIZE_BTC;
    const pnl = notionalUsd * pnlPct;

    logger.info({ id, side, entryPrice, exitPrice, pnl, reason }, "SignalTracker: signal closed");

    await dbService.saveTrade({
      id,
      symbol: "BTCUSDT",
      side,
      entryPrice,
      exitPrice,
      size: VIRTUAL_SIZE_BTC,
      leverage,
      pnl,
      pnlPct,
      openedAt: openedAt.toISOString(),
      closedAt: new Date().toISOString(),
      closeReason: reason,
    }, "Signal");

    const emoji = reason === "tp" ? "✅" : "❌";
    const pnlStr = `${pnl >= 0 ? "+" : ""}$${pnl.toFixed(2)}`;
    const pnlPctStr = `${pnlPct >= 0 ? "+" : ""}${(pnlPct * 100).toFixed(1)}%`;

    telegramService.notifyChannel(
      `${emoji} <b>ЗАКРЫТИЕ ${side === "long" ? "LONG" : "SHORT"}</b> — ${reason === "tp" ? "Take Profit 🎯" : "Stop Loss 🛑"}\n` +
      `━━━━━━━━━━━━━━━━━━━\n` +
      `💰 Цена выхода: <b>$${exitPrice.toFixed(1)}</b>\n` +
      `📊 P&amp;L: <b>${pnlStr}</b> (${pnlPctStr} на плече ${leverage}x)\n` +
      `━━━━━━━━━━━━━━━━━━━\n` +

  }
}

class BotEngine {
  private running = false;

        suggestedTp: s.suggestedTp,

      });

    }

    this.riskPct = config.riskPct;
    this.startedAt = new Date();
    this.tradesExecuted = 0;

    this.priceInterval = setInterval(() => {

      const positions = riskManager.getPositions();
      await dbService.saveSnapshot(

        stats.dailyPnl,

      await dbService.syncOpenPositions(positions);
    }, SNAPSHOT_INTERVAL_MS);

    const exchanges = aggregator.getActiveExchanges().join(", ");
    logger.info({ mode: config.mode, maxLeverage: config.maxLeverage, exchanges }, "Bot started");

    const price = aggregator.getPrimaryPrice();
    if (price === 0) return;

    const openPositions = riskManager.getPositions();
    const connector = aggregator.getBestConnectorForTrading();

    for (const pos of openPositions) {
      const hitSl = pos.side === "long" ? price <= pos.stopLoss : price >= pos.stopLoss;
      const hitTp = pos.side === "long" ? price >= pos.takeProfit : price <= pos.takeProfit;

      if (hitSl || hitTp) {
        const reason = hitTp ? "tp" : "sl";
        const trade = riskManager.closePosition(pos.id, price, reason);
        if (trade) {

          }

    if (!signal.allGatesPassed) return;

    this.lastTradeAt = now;

    telegramService.publishTrade({
      action: "open",
      side: signal.direction,
      price,
      size: sizeInBtc,
      leverage,
      exchange: connector.name,
      dataExchanges: aggregator.getActiveExchangeNames().join(" + "),

    const tpPct = Math.abs((newPos.takeProfit - price) / price * 100).toFixed(2);
    const prefix = this.mode === "paper" ? "[PAPER]" : "[LIVE]";

    telegramService.notify(

      try {
        await connector.placeOrder({

          side: signal.direction === "long" ? "Buy" : "Sell",

        this.tradesExecuted--;
      }
    }

import { logger } from "../lib/logger";

export interface Position {
  id: string;
  symbol: string;
  side: "long" | "short";
  entryPrice: number;
  currentPrice: number;
  size: number;
  leverage: number;
  pnl: number;
  pnlPct: number;
  stopLoss: number;
  takeProfit: number;
  openedAt: string;
}

export interface ClosedTrade {
  id: string;
  symbol: string;
  side: "long" | "short";
  entryPrice: number;
  exitPrice: number;
  size: number;
  leverage: number;
  pnl: number;
  pnlPct: number;
  openedAt: string;
  closedAt: string;
  closeReason: string;
}

const DAILY_LOSS_LIMIT = 0.03;
const COOLDOWN_MS = 60 * 60 * 1000;

class RiskManager {
  private positions = new Map<string, Position>();
  private closedToday: ClosedTrade[] = [];
  private accountBalance = 10000;
  private peakEquity = 10000;
  private circuitBreakerTripped = false;
  private cooldownUntil: Date | null = null;
  private pnlHistory: number[] = [];
  private onTradeClose: ((trade: ClosedTrade) => void) | null = null;

  setTradeCloseCallback(cb: (trade: ClosedTrade) => void) {
    this.onTradeClose = cb;
  }

  setAccountBalance(balance: number) {
    if (balance > 0) {
      this.accountBalance = balance;
      if (balance > this.peakEquity) this.peakEquity = balance;
    }
  }

  getAccountBalance(): number { return this.accountBalance; }

  checkCircuitBreaker(): boolean {
    if (this.cooldownUntil && Date.now() >= this.cooldownUntil.getTime()) {
      this.cooldownUntil = null;
      this.circuitBreakerTripped = false;
      logger.info("Circuit breaker reset after cooldown");
    }
    if (this.circuitBreakerTripped) return true;

    const dailyPnlPct = this.getDailyPnl() / this.accountBalance;
    if (dailyPnlPct <= -DAILY_LOSS_LIMIT) {
      this.circuitBreakerTripped = true;
      this.cooldownUntil = new Date(Date.now() + COOLDOWN_MS);
      logger.warn({ dailyPnlPct }, "Circuit breaker TRIPPED — 3% daily loss exceeded");
    }
    return this.circuitBreakerTripped;
  }

  openPosition(pos: Position): void {
    this.positions.set(pos.id, pos);
    logger.info({ id: pos.id, side: pos.side, price: pos.entryPrice }, "Position opened");
  }

  updatePositionPrices(currentPrice: number): void {
    for (const [id, pos] of this.positions) {
      const priceDiff = currentPrice - pos.entryPrice;
      const pnl = (pos.side === "long" ? priceDiff : -priceDiff) * pos.size;
      const pnlPct = pnl / (pos.entryPrice * pos.size / pos.leverage);
      this.positions.set(id, { ...pos, currentPrice, pnl, pnlPct });
    }
  }

  closePosition(id: string, exitPrice: number, reason = "manual"): ClosedTrade | null {
    const pos = this.positions.get(id);
    if (!pos) return null;
    this.positions.delete(id);

    const priceDiff = exitPrice - pos.entryPrice;
    const pnl = (pos.side === "long" ? priceDiff : -priceDiff) * pos.size;
    const pnlPct = pnl / (pos.entryPrice * pos.size / pos.leverage);

    const trade: ClosedTrade = {
      id: pos.id,
      symbol: pos.symbol,
      side: pos.side,
      entryPrice: pos.entryPrice,
      exitPrice,
      size: pos.size,
      leverage: pos.leverage,
      pnl,
      pnlPct,
      openedAt: pos.openedAt,
      closedAt: new Date().toISOString(),
      closeReason: reason,
    };

    this.closedToday.unshift(trade);
    if (this.closedToday.length > 200) this.closedToday = this.closedToday.slice(0, 200);
    this.pnlHistory.push(pnl);
    if (this.pnlHistory.length > 100) this.pnlHistory = this.pnlHistory.slice(-100);

    this.accountBalance += pnl;
    if (this.accountBalance > this.peakEquity) this.peakEquity = this.accountBalance;

    logger.info({ id: pos.id, pnl, reason }, "Position closed");
    if (this.onTradeClose) this.onTradeClose(trade);
    return trade;
  }

  loadClosedTrades(dbTrades: Array<{
    id: string;
    symbol: string;
    side: string;
    entryPrice: string | number;
    exitPrice: string | number;
    size: string | number;
    leverage: number;
    pnl: string | number;
    pnlPct: string | number;
    openedAt: string | null;
    closedAt: string | null;
    closeReason: string | null;
  }>): void {
    if (dbTrades.length === 0) return;
    const restored: ClosedTrade[] = dbTrades.map(t => ({
      id: t.id,
      symbol: t.symbol,
      side: t.side as "long" | "short",
      entryPrice: parseFloat(String(t.entryPrice)),
      exitPrice: parseFloat(String(t.exitPrice)),
      size: parseFloat(String(t.size)),
      leverage: t.leverage,
      pnl: parseFloat(String(t.pnl)),
      pnlPct: parseFloat(String(t.pnlPct)),
      openedAt: String(t.openedAt ?? ""),
      closedAt: String(t.closedAt ?? ""),
      closeReason: t.closeReason ?? "manual",
    }));
    this.closedToday = restored;
    this.pnlHistory = restored.map(t => t.pnl).slice(-100);
    logger.info({ count: restored.length }, "Restored today closed trades from DB into risk manager");
  }

  resetDailyTrades(): void {
    this.closedToday = [];
    this.circuitBreakerTripped = false;
    this.cooldownUntil = null;
  }

  getDailyPnl(): number {
    return this.closedToday.reduce((s, t) => s + t.pnl, 0);
  }

  getPositions(): Position[] {
    return Array.from(this.positions.values());
  }

  getClosedTrades(): ClosedTrade[] {
    return [...this.closedToday];
  }

  getStats() {
    const wins = this.closedToday.filter(t => t.pnl > 0);
    const losses = this.closedToday.filter(t => t.pnl < 0);
    const dailyPnl = this.getDailyPnl();
    const maxDrawdown = this.peakEquity > 0
      ? (this.peakEquity - this.accountBalance) / this.peakEquity : 0;

    let sharpe = 0;
    if (this.pnlHistory.length > 2) {
      const avg = this.pnlHistory.reduce((s, v) => s + v, 0) / this.pnlHistory.length;
      const variance = this.pnlHistory.reduce((s, v) => s + (v - avg) ** 2, 0) / this.pnlHistory.length;
      sharpe = Math.sqrt(variance) > 0 ? avg / Math.sqrt(variance) : 0;
    }

    return {
      dailyPnl: parseFloat(dailyPnl.toFixed(4)),
      dailyPnlPct: parseFloat((dailyPnl / this.accountBalance).toFixed(6)),
      maxDrawdown: parseFloat(maxDrawdown.toFixed(6)),
      winRate: this.closedToday.length > 0
        ? parseFloat((wins.length / this.closedToday.length).toFixed(4)) : 0,
      totalTrades: this.closedToday.length,
      winningTrades: wins.length,
      losingTrades: losses.length,
      avgWin: wins.length > 0
        ? parseFloat((wins.reduce((s, t) => s + t.pnl, 0) / wins.length).toFixed(4)) : 0,
      avgLoss: losses.length > 0
        ? parseFloat((losses.reduce((s, t) => s + t.pnl, 0) / losses.length).toFixed(4)) : 0,
      sharpeIntraday: parseFloat(sharpe.toFixed(4)),
      circuitBreakerTripped: this.circuitBreakerTripped,
      cooldownUntil: this.cooldownUntil?.toISOString() ?? null,
    };
  }
}

function startOfDayMsk(msk: Date): Date {
  const d = new Date(msk);
  d.setUTCHours(0, 0, 0, 0);
  return new Date(d.getTime() - 3 * 60 * 60 * 1000);

  private lastDaily = "";
  private lastWeekly = "";
  private lastMonthly = "";

  start() {
    this.intervalId = setInterval(() => void this.check(), 60 * 1000);
    logger.info("Stats scheduler started (daily 21:00, weekly Sun 12:00, monthly 1st 10:00 MSK)");
  }

  stop() {
    if (this.intervalId) clearInterval(this.intervalId);

  private async check() {
    const msk = mskNow();
    const dayOfWeek = msk.getUTCDay();
    const dayOfMonth = msk.getUTCDate();
    const hour = msk.getUTCHours();
    const minute = msk.getUTCMinutes();
    const dateKey = msk.toISOString().slice(0, 10);

    if (hour === 21 && minute === 0 && this.lastDaily !== dateKey) {
      this.lastDaily = dateKey;
      await this.sendDailyReport(msk);
    }

    if (dayOfWeek === 0 && hour === 12 && minute === 0 && this.lastWeekly !== dateKey) {
      this.lastWeekly = dateKey;
      await this.sendWeeklyReport(msk);
    }

    if (dayOfMonth === 1 && hour === 10 && minute === 0 && this.lastMonthly !== dateKey) {
      this.lastMonthly = dateKey;
      await this.sendMonthlyReport(msk);
    }
  }

  confidence: number;
  bullishMatches: number;
  bearishMatches: number;
  topMatches: PatternMatch[];
  currentReturns: number[];
  updatedAt: string;
}

const PATTERN_WINDOW = 7;
const FORWARD_WINDOW = 3;
const REFRESH_INTERVAL_MS = 6 * 60 * 60 * 1000;
const TOP_N = 10;

function pearson(a: number[], b: number[]): number {
  const n = Math.min(a.length, b.length);
  if (n === 0) return 0;
  const meanA = a.slice(0, n).reduce((s, v) => s + v, 0) / n;
  const meanB = b.slice(0, n).reduce((s, v) => s + v, 0) / n;
  let num = 0, devA = 0, devB = 0;
  for (let i = 0; i < n; i++) {
    const da = a[i] - meanA;
    const db = b[i] - meanB;
    num += da * db;
    devA += da * da;
    devB += db * db;
  }
  const denom = Math.sqrt(devA * devB);
  return denom === 0 ? 0 : num / denom;
}

function toReturns(candles: DailyCandle[]): number[] {
  return candles.map(c => (c.close - c.open) / c.open);
}

function normalize(returns: number[]): number[] {
  const mean = returns.reduce((s, v) => s + v, 0) / returns.length;
  const std = Math.sqrt(returns.reduce((s, v) => s + (v - mean) ** 2, 0) / returns.length);
  if (std === 0) return returns.map(() => 0);
  return returns.map(v => (v - mean) / std);
}

class PatternService {
  private candles: DailyCandle[] = [];
  private lastResult: PatternResult | null = null;
  private intervalId: ReturnType<typeof setInterval> | null = null;
  private lastFetch = 0;

  async start() {
    await this.fetchAndCompute();
    this.intervalId = setInterval(() => void this.fetchAndCompute(), REFRESH_INTERVAL_MS);
    logger.info("Pattern service started (7-day pattern matching vs full BTC history from 2013)");
  }

  stop() {
    if (this.intervalId) clearInterval(this.intervalId);
  }

  private async fetchFromCryptoCompare(): Promise<DailyCandle[]> {
    // CryptoCompare histoday — free, no API key, up to 2000 days of daily OHLCV
    // Covers: 2019 accumulation, 2020 crash+recovery, 2021 bull ATH, 2022 bear, 2023 recovery, 2024-25 bull
    const url = "https://min-api.cryptocompare.com/data/v2/histoday?fsym=BTC&tsym=USD&limit=2000&toTs=-1";
    const resp = await fetch(url, {
      signal: AbortSignal.timeout(15000),
      headers: { "Accept": "application/json" },
    });
    if (!resp.ok) throw new Error(`CryptoCompare HTTP ${resp.status}`);
    const json = await resp.json() as {
      Response: string;
      Data: { Data: { time: number; open: number; high: number; low: number; close: number }[] };
    };
    if (json.Response !== "Success") throw new Error(`CryptoCompare error: ${json.Response}`);
    const rows = json.Data.Data;
    if (!Array.isArray(rows) || rows.length < 30) throw new Error("CryptoCompare returned insufficient data");

    return rows
      .filter(r => r.close > 0)
      .map(r => ({
        ts: r.time * 1000,
        open: r.open,
        high: r.high,
        low: r.low,
        close: r.close,
      }));
  }

  private async fetchCandles(): Promise<DailyCandle[]> {
    try {
      const candles = await this.fetchFromCryptoCompare();
      return candles;
    } catch (e) {
      logger.warn({ err: e }, "Pattern service: CryptoCompare fetch failed, trying Bybit connector");
    }
    try {
      const candles = await aggregator.fetchDailyKlines(200);
      logger.info({ candles: candles.length }, "Pattern service: fetched daily candles from Bybit");
      return candles;
    } catch (e2) {
      logger.warn({ err: e2 }, "Pattern service: all kline sources failed");
      return [];
    }
  }

  private async fetchAndCompute() {
    const fresh = await this.fetchCandles();
    if (fresh.length >= PATTERN_WINDOW + FORWARD_WINDOW + 10) {
      this.candles = fresh;
      this.lastFetch = Date.now();
      this.lastResult = this.compute();
      logger.info({ candles: this.candles.length, signal: this.lastResult?.signal.toFixed(3) }, "Pattern matching updated");
    }
  }

  private compute(): PatternResult {
    const candles = this.candles;
    const total = candles.length;

    const completedCandles = candles.slice(0, total - 1);

    const currentSlice = completedCandles.slice(-PATTERN_WINDOW);
    const currentReturns = toReturns(currentSlice);
    const currentNorm = normalize(currentReturns);

    const matches: { similarity: number; forwardPct: number; startIdx: number }[] = [];

    for (let i = 0; i <= completedCandles.length - PATTERN_WINDOW - FORWARD_WINDOW; i++) {
      const window = completedCandles.slice(i, i + PATTERN_WINDOW);
      const winReturns = toReturns(window);
      const winNorm = normalize(winReturns);
      const sim = pearson(currentNorm, winNorm);

      const forwardSlice = completedCandles.slice(i + PATTERN_WINDOW, i + PATTERN_WINDOW + FORWARD_WINDOW);
      const forwardPct = forwardSlice.length > 0
        ? (forwardSlice[forwardSlice.length - 1].close - forwardSlice[0].open) / forwardSlice[0].open * 100
        : 0;

      matches.push({ similarity: sim, forwardPct, startIdx: i });
    }

    matches.sort((a, b) => b.similarity - a.similarity);
    const top = matches.slice(0, TOP_N);

    const topMatches: PatternMatch[] = top.map(m => {
      const window = completedCandles.slice(m.startIdx, m.startIdx + PATTERN_WINDOW);
      const forwardSlice = completedCandles.slice(m.startIdx + PATTERN_WINDOW, m.startIdx + PATTERN_WINDOW + FORWARD_WINDOW);

      const fwdPct = forwardSlice.length > 0
        ? (forwardSlice[forwardSlice.length - 1].close - forwardSlice[0].open) / forwardSlice[0].open * 100
        : 0;

      return {
        startDate: new Date(window[0].ts).toISOString().slice(0, 10),
        endDate: new Date(window[window.length - 1].ts).toISOString().slice(0, 10),
        similarity: parseFloat(m.similarity.toFixed(4)),
        forwardPct: parseFloat(fwdPct.toFixed(4)),
        outcome: fwdPct > 0.5 ? "bullish" : fwdPct < -0.5 ? "bearish" : "neutral",
        currentReturns: currentReturns.map(v => parseFloat(v.toFixed(6))),
        historicalReturns: toReturns(window).map(v => parseFloat(v.toFixed(6))),
        forwardReturns: toReturns(forwardSlice).map(v => parseFloat(v.toFixed(6))),
      };
    });

    const bullishMatches = topMatches.filter(m => m.outcome === "bullish").length;
    const bearishMatches = topMatches.filter(m => m.outcome === "bearish").length;

    const weightedSignal = top.reduce((sum, m) => {
      const direction = m.forwardPct > 0 ? 1 : m.forwardPct < 0 ? -1 : 0;
      return sum + direction * Math.abs(m.similarity);
    }, 0);
    const totalWeight = top.reduce((sum, m) => sum + Math.abs(m.similarity), 0);
    const signal = totalWeight > 0 ? Math.max(-1, Math.min(1, weightedSignal / totalWeight)) : 0;

    const avgSim = top.length > 0 ? top.reduce((s, m) => s + Math.abs(m.similarity), 0) / top.length : 0;
    const agreementRatio = top.length > 0 ? Math.max(bullishMatches, bearishMatches) / top.length : 0;
    const confidence = avgSim * agreementRatio;

    return {
      signal: parseFloat(signal.toFixed(4)),
      confidence: parseFloat(confidence.toFixed(4)),
      bullishMatches,
      bearishMatches,
      topMatches,
      currentReturns: currentReturns.map(v => parseFloat(v.toFixed(6))),
      updatedAt: new Date().toISOString(),
    };
  }

  getResult(): PatternResult | null {
    return this.lastResult;
  }

  getSignal(): number {
    return this.lastResult?.signal ?? 0;
  }

  getConfidence(): number {
    return this.lastResult?.confidence ?? 0;
  }

  isReady(): boolean {
    return this.lastResult !== null;
  }
}

export interface Candle {
  time: number;
  open: number;
  high: number;
  low: number;
  close: number;
  volume: number;
  buyVolume: number;
  sellVolume: number;
}

const MAX_CANDLES = 100;

class CandleService {
  private candles: Candle[] = [];
  private currentCandle: Candle | null = null;
  private processedTradeIds = new Set<string>();
  private intervalHandle: ReturnType<typeof setInterval> | null = null;

  private vwapNumerator = 0;
  private vwapDenominator = 0;
  private vwapDayStart = 0;

  start() {
    this.intervalHandle = setInterval(() => this.tick(), 2000);
  }

  stop() {
    if (this.intervalHandle) clearInterval(this.intervalHandle);
  }

  private tick() {
    const trades = aggregator.getAllTrades().filter(t => !this.processedTradeIds.has(t.id));
    for (const t of trades) {
      this.processedTradeIds.add(t.id);
      this.addTick(t.price, t.size, t.side as "buy" | "sell", t.timestamp);
    }
    if (this.processedTradeIds.size > 20000) {
      const arr = Array.from(this.processedTradeIds);
      this.processedTradeIds = new Set(arr.slice(-10000));
    }
  }

  private addTick(price: number, size: number, side: "buy" | "sell", timestamp: number) {
    const dayStartTs = new Date(timestamp).setUTCHours(0, 0, 0, 0);
    if (dayStartTs !== this.vwapDayStart) {
      this.vwapDayStart = dayStartTs;
      this.vwapNumerator = 0;
      this.vwapDenominator = 0;
    }
    this.vwapNumerator += price * size;
    this.vwapDenominator += size;

    const minute = Math.floor(timestamp / 60000) * 60000;
    if (!this.currentCandle || this.currentCandle.time !== minute) {
      if (this.currentCandle) {
        this.candles.push(this.currentCandle);
        if (this.candles.length > MAX_CANDLES) this.candles.shift();
      }
      this.currentCandle = {
        time: minute,
        open: price, high: price, low: price, close: price,
        volume: size,
        buyVolume: side === "buy" ? size : 0,
        sellVolume: side === "sell" ? size : 0,
      };
    } else {
      if (price > this.currentCandle.high) this.currentCandle.high = price;
      if (price < this.currentCandle.low) this.currentCandle.low = price;
      this.currentCandle.close = price;
      this.currentCandle.volume += size;
      if (side === "buy") this.currentCandle.buyVolume += size;
      else this.currentCandle.sellVolume += size;
    }
  }

  getCandles(): Candle[] {
    const all = [...this.candles];
    if (this.currentCandle) all.push({ ...this.currentCandle });
    return all;
  }

  getCandleCount(): number {
    return this.candles.length + (this.currentCandle ? 1 : 0);
  }

  getVwap(): number {
    if (this.vwapDenominator === 0) return 0;
    return this.vwapNumerator / this.vwapDenominator;
  }

  getAtr(period = 14): number {
    const candles = this.getCandles();
    if (candles.length < 2) return 0;
    const trs: number[] = [];
    for (let i = 1; i < candles.length; i++) {
      const h = candles[i].high;
      const l = candles[i].low;
      const pc = candles[i - 1].close;
      trs.push(Math.max(h - l, Math.abs(h - pc), Math.abs(l - pc)));
    }
    const lookback = Math.min(period, trs.length);
    return trs.slice(-lookback).reduce((s, v) => s + v, 0) / lookback;
  }

  getEma(period: number): number {
    const candles = this.getCandles();
    if (candles.length === 0) return 0;
    const closes = candles.map(c => c.close);
    if (closes.length < period) {
      return closes.reduce((s, v) => s + v, 0) / closes.length;
    }
    const k = 2 / (period + 1);
    let ema = closes.slice(0, period).reduce((s, v) => s + v, 0) / period;
    for (let i = period; i < closes.length; i++) {
      ema = closes[i] * k + ema * (1 - k);
    }
    return ema;
  }
}

import { aggregator } from "./exchanges/aggregator";

export interface CvdPoint {
  timestamp: string;
  cvd: number;
  delta: number;
  price: number;
}

const MAX_POINTS = 120;

class CvdService {
  private points: CvdPoint[] = [];
  private processedTradeIds = new Set<string>();
  private lastProcessed = 0;
  private intervalHandle: ReturnType<typeof setInterval> | null = null;

  start() {
    this.intervalHandle = setInterval(() => this.tick(), 1000);
  }

  stop() {
    if (this.intervalHandle) clearInterval(this.intervalHandle);
    this.intervalHandle = null;
  }

  private tick() {
    const now = Date.now();
    const windowStart = this.lastProcessed || now - 2000;

    const trades = aggregator.getPrimaryTrades().filter(
      t => t.timestamp >= windowStart && !this.processedTradeIds.has(t.id)
    );

    if (trades.length === 0) return;

    let delta = 0;
    let lastPrice = aggregator.getPrimaryPrice();

    for (const t of trades) {
      delta += t.side === "buy" ? t.size : -t.size;
      this.processedTradeIds.add(t.id);
      lastPrice = t.price;
    }

    const prevCvd = this.points.length > 0 ? this.points[this.points.length - 1].cvd : 0;

    this.points.push({
      timestamp: new Date(now).toISOString(),
      cvd: prevCvd + delta,
      delta,
      price: lastPrice,
    });

    if (this.points.length > MAX_POINTS) {
      this.points = this.points.slice(-MAX_POINTS);
    }

    this.lastProcessed = now;

    if (this.processedTradeIds.size > 10000) {
      this.processedTradeIds.clear();
    }
  }

  getPoints(): CvdPoint[] {
    return [...this.points];
  }

  getCurrentCvd(): number {
    return this.points.length > 0 ? this.points[this.points.length - 1].cvd : 0;
  }

  getTrend(): "bullish" | "bearish" | "neutral" {
    if (this.points.length < 10) return "neutral";
    const recent = this.points.slice(-10);
    const first = recent[0].cvd;
    const last = recent[recent.length - 1].cvd;
    const delta = last - first;
    const threshold = Math.abs(first) * 0.02 + 10;
    if (delta > threshold) return "bullish";
    if (delta < -threshold) return "bearish";
    return "neutral";
  }
}

function caRequest(path: string): Promise<unknown> {
  return new Promise((resolve, reject) => {
    const options = {
      hostname: BASE,
      path,
      method: "GET",
      headers: { "api_key": API_KEY, "Accept": "application/json" },
    };
    const req = https.request(options, (res) => {
      let data = "";
      res.on("data", chunk => data += chunk);

  // ── Liquidation history ──────────────────────────────────────

  private async fetchLiqHistory(from: number, to: number, interval: string): Promise<LiqCandle[]> {

  private async refreshMap() {
    if (this.fetchingMap) return;
    this.fetchingMap = true;
    try {
      const to = Math.floor(Date.now() / 1000);
      const from = to - 7 * 86400; // 7 days of hourly data

        caRequest(`/v1/ohlcv-history?symbols=${SYMBOL}&interval=1hour&from=${from}&to=${to}`),
        caRequest(`/v1/open-interest-history?symbols=${SYMBOL}&interval=1hour&from=${from}&to=${to}`),
      ]) as [
        Array<{ symbol: string; history: OhlcvCandle[] }>,
        Array<{ symbol: string; history: OiCandle[] }>
      ];

      if (ohlcvRaw[0]?.history?.length) this.cacheOhlcv = ohlcvRaw[0].history;
      if (oiRaw[0]?.history?.length) this.cacheOi = oiRaw[0].history;
      this.cacheMapAt = Date.now();
      this.cachedMap = this.computeMap();
    } catch (e) { logger.warn({ err: e }, "Coinalyze map refresh failed"); }
    finally { this.fetchingMap = false; }
  }

  private computeMap(): LiquidationMap | null {
    if (!this.cacheOhlcv.length) return null;

    const oiByTime = new Map<number, number>();
    this.cacheOi.forEach(c => oiByTime.set(c.t, c.c));

    // Determine price range for buckets
    const allPrices = this.cacheOhlcv.flatMap(c => [c.h, c.l]);
    const globalMin = Math.min(...allPrices) * 0.97;
    const globalMax = Math.max(...allPrices) * 1.03;
    const BUCKET_SIZE = 200; // $200 buckets
    const bucketCount = Math.ceil((globalMax - globalMin) / BUCKET_SIZE);

    const longBuckets = new Float64Array(bucketCount);
    const shortBuckets = new Float64Array(bucketCount);

    // Get current price (last close)
    const lastCandle = this.cacheOhlcv[this.cacheOhlcv.length - 1];
    const currentPrice = lastCandle?.c ?? 80000;

    // Build OI deltas: positive delta = new net positions opened
    for (let i = 1; i < this.cacheOhlcv.length; i++) {
      const c = this.cacheOhlcv[i]!;
      const prev = this.cacheOhlcv[i - 1]!;

      // Entry price approximation: VWAP of candle
      const entryPrice = (c.o + c.h + c.l + c.c) / 4;

      // Volume-based weight (more volume = more positions at this price)
      const volumeWeight = c.v;

      // OI at this candle — use as weight multiplier
      const oi = oiByTime.get(c.t) ?? oiByTime.get(prev.t) ?? 1;

      // Buy volume fraction → estimate long bias
      const buyFrac = c.v > 0 ? c.bv / c.v : 0.5;
      const sellFrac = 1 - buyFrac;

      // Position weight in USD
      const posWeight = volumeWeight * entryPrice * 0.01; // scale factor

      for (const tier of LEVERAGE_TIERS) {
        const tierWeight = tier.weight * posWeight * (oi / 100000);

        // Long liquidation price: price drops 0.9/lev from entry
        const longLiqPrice = entryPrice * (1 - 0.9 / tier.lev);
        // Short liquidation price: price rises 0.9/lev from entry
        const shortLiqPrice = entryPrice * (1 + 0.9 / tier.lev);

        // Only add if within range
        const addToBucket = (price: number, value: number, arr: Float64Array) => {
          if (price < globalMin || price > globalMax) return;
          const idx = Math.floor((price - globalMin) / BUCKET_SIZE);
          if (idx >= 0 && idx < bucketCount) arr[idx] += value;
        };

        addToBucket(longLiqPrice, tierWeight * buyFrac, longBuckets);
        addToBucket(shortLiqPrice, tierWeight * sellFrac, shortBuckets);
      }
    }

    // Convert to array, normalize to USD millions for readability
    const buckets: LiqMapBucket[] = [];
    for (let i = 0; i < bucketCount; i++) {
      const price = globalMin + i * BUCKET_SIZE + BUCKET_SIZE / 2;
      const longLiqUsd = longBuckets[i]!;
      const shortLiqUsd = shortBuckets[i]!;
      const totalUsd = longLiqUsd + shortLiqUsd;
      if (totalUsd > 0) {
        buckets.push({ price, longLiqUsd, shortLiqUsd, totalUsd });
      }
    }

    return { buckets, currentPrice, priceRange: { min: globalMin, max: globalMax }, updatedAt: Date.now() };
  }

  getCachedMap(): LiquidationMap | null { return this.cachedMap; }

  async getLiquidationMap(): Promise<LiquidationMap | null> {
    if (!this.cachedMap || Date.now() - this.cacheMapAt > 5 * 60 * 1000) {
      await this.refreshMap();
    }
    return this.cachedMap;
  }
}

function cgRequest(path: string): Promise<Record<string, unknown>> {
  return new Promise((resolve, reject) => {
    const options = {
      hostname: BASE,
      path,
      method: "GET",
      headers: {
        "coinglassSecret": API_KEY,
        "Accept": "application/json",
      },
    };

    const req = https.request(options, (res) => {
      let data = "";
      res.on("data", chunk => data += chunk);
      res.on("end", () => {
        try {
          const parsed = JSON.parse(data) as Record<string, unknown>;
          if (parsed["code"] !== "0" && parsed["code"] !== 0) {
            reject(new Error(`CoinGlass error: ${parsed["msg"] ?? parsed["code"]}`));
          } else {
            resolve(parsed["data"] as Record<string, unknown>);
          }
        } catch (e) { reject(e); }
      });
    });
    req.on("error", reject);
    req.end();
  });
}

import type { ExchangeLiquidation } from "./exchanges/types";

const MAX_EVENTS = 200;
const BUCKET_SIZE_USD = 100;
const WINDOW_MS = 30 * 60 * 1000;

export interface LiqBucket {
  price: number;
  longUsd: number;
  shortUsd: number;
  totalUsd: number;
}

export interface LiqEvent {
  exchange: string;
  side: "long" | "short";
  price: number;
  size: number;
  valueUsd: number;
  timestamp: number;
}

class LiquidationService {
  private events: LiqEvent[] = [];

  addEvent(liq: ExchangeLiquidation) {
    this.events.unshift({
      exchange: liq.exchange,
      side: liq.side,
      price: liq.price,
      size: liq.size,
      valueUsd: liq.valueUsd,
      timestamp: liq.timestamp,
    });
    if (this.events.length > MAX_EVENTS) this.events.length = MAX_EVENTS;
  }

  getRecentEvents(limit = 50): LiqEvent[] {
    return this.events.slice(0, limit);
  }

  getHeatmapBuckets(): LiqBucket[] {
    const cutoff = Date.now() - WINDOW_MS;
    const recent = this.events.filter(e => e.timestamp >= cutoff);
    if (!recent.length) return [];

    const minPrice = Math.min(...recent.map(e => e.price));
    const maxPrice = Math.max(...recent.map(e => e.price));
    const bucketCount = Math.min(60, Math.ceil((maxPrice - minPrice) / BUCKET_SIZE_USD) + 1);
    const actualBucketSize = bucketCount > 1 ? (maxPrice - minPrice) / (bucketCount - 1) : BUCKET_SIZE_USD;
    const basePrice = minPrice;

    const map = new Map<number, LiqBucket>();

    for (const e of recent) {
      const idx = actualBucketSize > 0
        ? Math.round((e.price - basePrice) / actualBucketSize)
        : 0;
      const bucketPrice = Math.round((basePrice + idx * actualBucketSize) / 10) * 10;

      if (!map.has(bucketPrice)) {
        map.set(bucketPrice, { price: bucketPrice, longUsd: 0, shortUsd: 0, totalUsd: 0 });
      }
      const b = map.get(bucketPrice)!;
      if (e.side === "long") b.longUsd += e.valueUsd;
      else b.shortUsd += e.valueUsd;
      b.totalUsd += e.valueUsd;
    }

    return Array.from(map.values()).sort((a, b) => a.price - b.price);
  }

  getSummary() {
    const now = Date.now();
    const inWindow = (ms: number) => this.events.filter(e => e.timestamp >= now - ms);

    const calc = (evts: LiqEvent[]) => ({
      longUsd: evts.filter(e => e.side === "long").reduce((s, e) => s + e.valueUsd, 0),
      shortUsd: evts.filter(e => e.side === "short").reduce((s, e) => s + e.valueUsd, 0),
      count: evts.length,
    });

    return {
      last5m: calc(inWindow(5 * 60 * 1000)),
      last15m: calc(inWindow(15 * 60 * 1000)),
      last30m: calc(inWindow(30 * 60 * 1000)),
      last1h: calc(inWindow(60 * 60 * 1000)),
    };
  }
}

import { logger } from "../lib/logger";

export interface BybitTicker {

export interface BybitOrderbookLevel {

  size: number;
}

export interface BybitOrderbook {
  symbol: string;
  bids: BybitOrderbookLevel[];
  asks: BybitOrderbookLevel[];

export interface BybitTrade {
  id: string;
  price: number;
  size: number;

const BYBIT_WS = "wss://stream.bybit.com/v5/public/linear";
const SYMBOL = "BTCUSDT";
const MAX_TRADES = 200;
const MAX_BOOK_LEVELS = 25;

class BybitService {
  private ws: WebSocket | null = null;
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
  private pingTimer: ReturnType<typeof setInterval> | null = null;
  private reconnectDelay = 1000;

  ticker: BybitTicker = {
    symbol: SYMBOL,
    lastPrice: 0,
    price24hPcnt: 0,
    volume24h: 0,
    highPrice24h: 0,
    lowPrice24h: 0,
    prevPrice24h: 0,
    openInterest: 0,
    fundingRate: 0,
    nextFundingTime: 0,
  };

  orderbook: BybitOrderbook = {
    symbol: SYMBOL,
    bids: [],
    asks: [],
    updatedAt: Date.now(),
  };

  trades: BybitTrade[] = [];

  private bidsMap = new Map<string, number>();
  private asksMap = new Map<string, number>();

  connect() {
    if (this.ws) return;
    logger.info("Connecting to Bybit WebSocket...");
    this.ws = new WebSocket(BYBIT_WS);

    this.ws.on("open", () => {
      logger.info("Bybit WebSocket connected");
      this.reconnectDelay = 1000;
      this.subscribe();
      this.startPing();
    });

    this.ws.on("message", (data: Buffer) => {
      try {
        const msg = JSON.parse(data.toString());
        this.handleMessage(msg);
      } catch (e) {
        logger.warn({ err: e }, "Failed to parse Bybit message");
      }
    });

    this.ws.on("close", () => {
      logger.warn("Bybit WebSocket disconnected, reconnecting...");
      this.cleanup();
      this.scheduleReconnect();
    });

    this.ws.on("error", (err) => {
      logger.error({ err }, "Bybit WebSocket error");
      this.ws?.terminate();
    });
  }

  private subscribe() {
    const topics = [
      `tickers.${SYMBOL}`,
      `orderbook.50.${SYMBOL}`,
      `publicTrade.${SYMBOL}`,
    ];
    this.ws?.send(JSON.stringify({ op: "subscribe", args: topics }));
  }

  private startPing() {
    this.pingTimer = setInterval(() => {
      if (this.ws?.readyState === WebSocket.OPEN) {
        this.ws.send(JSON.stringify({ op: "ping" }));
      }
    }, 20000);
  }

  private cleanup() {
    if (this.pingTimer) clearInterval(this.pingTimer);
    this.pingTimer = null;
    this.ws = null;
  }

  private scheduleReconnect() {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
    this.reconnectTimer = setTimeout(() => {
      this.reconnectTimer = null;
      this.connect();
    }, this.reconnectDelay);
    this.reconnectDelay = Math.min(this.reconnectDelay * 2, 30000);
  }

  private handleMessage(msg: Record<string, unknown>) {
    if (msg["op"] === "pong" || msg["op"] === "subscribe") return;

    const topic = msg["topic"] as string | undefined;
    if (!topic) return;

    const data = msg["data"];

    if (topic === `tickers.${SYMBOL}`) {
      this.handleTicker(data as Record<string, string>);
    } else if (topic.startsWith("orderbook.")) {
      const type = msg["type"] as string;
      this.handleOrderbook(data as Record<string, unknown>, type);
    } else if (topic === `publicTrade.${SYMBOL}`) {
      this.handleTrades(data as Array<Record<string, string>>);
    }
  }

  private handleTicker(data: Record<string, string>) {
    const price = parseFloat(data["lastPrice"] ?? "0");
    if (price > 0) this.ticker.lastPrice = price;

    const pct = parseFloat(data["price24hPcnt"] ?? "");
    if (!isNaN(pct)) this.ticker.price24hPcnt = pct;

    const vol = parseFloat(data["volume24h"] ?? "");
    if (!isNaN(vol)) this.ticker.volume24h = vol;

    const high = parseFloat(data["highPrice24h"] ?? "");
    if (!isNaN(high)) this.ticker.highPrice24h = high;

    const low = parseFloat(data["lowPrice24h"] ?? "");
    if (!isNaN(low)) this.ticker.lowPrice24h = low;

    const prev = parseFloat(data["prevPrice24h"] ?? "");
    if (!isNaN(prev)) this.ticker.prevPrice24h = prev;

    const oi = parseFloat(data["openInterest"] ?? "");
    if (!isNaN(oi)) this.ticker.openInterest = oi;

    const fr = parseFloat(data["fundingRate"] ?? "");
    if (!isNaN(fr)) this.ticker.fundingRate = fr;

    const nft = parseInt(data["nextFundingTime"] ?? "0");
    if (nft > 0) this.ticker.nextFundingTime = nft;
  }

  private handleOrderbook(data: Record<string, unknown>, type: string) {
    if (type === "snapshot") {
      this.bidsMap.clear();
      this.asksMap.clear();
    }

    const bids = data["b"] as Array<[string, string]> | undefined;
    const asks = data["a"] as Array<[string, string]> | undefined;

    if (bids) {
      for (const [p, s] of bids) {
        const size = parseFloat(s);
        if (size === 0) this.bidsMap.delete(p);
        else this.bidsMap.set(p, size);
      }
    }

    if (asks) {
      for (const [p, s] of asks) {
        const size = parseFloat(s);
        if (size === 0) this.asksMap.delete(p);
        else this.asksMap.set(p, size);
      }
    }

    const bidsArr = Array.from(this.bidsMap.entries())
      .map(([p, s]) => ({ price: parseFloat(p), size: s }))
      .sort((a, b) => b.price - a.price)
      .slice(0, MAX_BOOK_LEVELS);

    const asksArr = Array.from(this.asksMap.entries())
      .map(([p, s]) => ({ price: parseFloat(p), size: s }))
      .sort((a, b) => a.price - b.price)
      .slice(0, MAX_BOOK_LEVELS);

    this.orderbook = {
      symbol: SYMBOL,
      bids: bidsArr,
      asks: asksArr,
      updatedAt: Date.now(),
    };
  }

  private handleTrades(data: Array<Record<string, string>>) {
    for (const t of data) {
      const trade: BybitTrade = {
        id: t["i"] ?? String(Date.now()),
        price: parseFloat(t["p"] ?? "0"),
        size: parseFloat(t["v"] ?? "0"),
        side: t["S"] === "Buy" ? "buy" : "sell",
        timestamp: parseInt(t["T"] ?? "0"),
      };
      this.trades.unshift(trade);
    }
    if (this.trades.length > MAX_TRADES) {
      this.trades = this.trades.slice(0, MAX_TRADES);
    }
  }
}

function restRequest(
  method: "GET" | "POST",
  path: string,
  params: Record<string, unknown> = {},
  apiKey: string,
  apiSecret: string,
): Promise<Record<string, unknown>> {
  return new Promise((resolve, reject) => {
    const timestamp = Date.now().toString();
    const recvWindow = "5000";
    let queryString = "";
    let body = "";

    if (method === "GET") {
      const entries = Object.entries(params).map(([k, v]) => `${k}=${v}`).join("&");
      queryString = entries;
    } else {
      body = JSON.stringify(params);
    }

    const signPayload = `${timestamp}${apiKey}${recvWindow}${method === "GET" ? queryString : body}`;
    const signature = hmacSha256(apiSecret, signPayload);

    const headers: Record<string, string> = {
      "X-BAPI-API-KEY": apiKey,
      "X-BAPI-TIMESTAMP": timestamp,
      "X-BAPI-RECV-WINDOW": recvWindow,
      "X-BAPI-SIGN": signature,
      "Content-Type": "application/json",
    };

    const url = new URL(`${REST_BASE}${path}${queryString ? "?" + queryString : ""}`);

    const options = {
      hostname: url.hostname,
      path: url.pathname + url.search,
      method,
      headers,
    };

    const req = https.request(options, (res) => {
      let data = "";
      res.on("data", (chunk) => (data += chunk));
      res.on("end", () => {
        try {
          const parsed = JSON.parse(data) as Record<string, unknown>;
          if (parsed["retCode"] !== 0) {
            reject(new Error(`Bybit API error ${parsed["retCode"]}: ${parsed["retMsg"]}`));
          } else {
            resolve((parsed["result"] ?? parsed) as Record<string, unknown>);
          }
        } catch (e) {
          reject(e);
        }
      });
    });

    req.on("error", reject);
    if (method === "POST" && body) req.write(body);
    req.end();
  });
}

export class BybitConnector implements IExchangeConnector {

  private bidsMap = new Map<string, number>();
  private asksMap = new Map<string, number>();
  private _orderbook: ExchangeOrderbook | null = null;

  private _liquidationCallbacks: Array<(liq: ExchangeLiquidation) => void> = [];

  private apiSecret = process.env["BYBIT_API_SECRET"] ?? "";

    if (this.ws) return;
    logger.info("Bybit: connecting to public WebSocket...");
    this.ws = new WebSocket(WS_PUBLIC);

    this.ws.on("open", () => {
      this._connected = true;
      this.reconnectDelay = 1000;
      this.subscribe();
      this.startPing();
      logger.info("Bybit: WebSocket connected");
    });

    this.ws.on("message", (data: Buffer) => {
      try { this.handleMessage(JSON.parse(data.toString())); }
      catch (e) { logger.warn({ err: e }, "Bybit: parse error"); }
    });

    this.ws.on("close", () => {
      this._connected = false;
      this.cleanup();
      this.scheduleReconnect();
      logger.warn("Bybit: WebSocket closed, reconnecting...");

    this.ws.on("error", (err) => {
      logger.error({ err }, "Bybit: WebSocket error");
      this.ws?.terminate();
    });
  }

  private subscribe() {
    this.ws?.send(JSON.stringify({
      op: "subscribe",
      args: [
        `tickers.${SYMBOL}`,
        `orderbook.50.${SYMBOL}`,
        `publicTrade.${SYMBOL}`,
        `allLiquidation.${SYMBOL}`,
      ],
    }));
  }

  private startPing() {
    this.pingTimer = setInterval(() => {
      if (this.ws?.readyState === WebSocket.OPEN)
        this.ws.send(JSON.stringify({ op: "ping" }));
    }, 20000);
  }

  private cleanup() {
    if (this.pingTimer) clearInterval(this.pingTimer);
    this.pingTimer = null;
    this.ws = null;
  }

  private scheduleReconnect() {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
    this.reconnectTimer = setTimeout(() => { this.reconnectTimer = null; this.connect(); }, this.reconnectDelay);
    this.reconnectDelay = Math.min(this.reconnectDelay * 2, 30000);
  }

  private handleMessage(msg: Record<string, unknown>) {
    if (msg["op"] === "pong" || msg["op"] === "subscribe") return;
    const topic = msg["topic"] as string | undefined;
    if (!topic) return;

    if (topic === `tickers.${SYMBOL}`) this.handleTicker(msg["data"] as Record<string, string>);
    else if (topic.startsWith("orderbook.")) this.handleOrderbook(msg["data"] as Record<string, unknown>, msg["type"] as string);
    else if (topic === `publicTrade.${SYMBOL}`) this.handleTrades(msg["data"] as Array<Record<string, string>>);
    else if (topic === `allLiquidation.${SYMBOL}`) this.handleLiquidation(msg["data"] as Record<string, unknown>);
  }

  private handleTicker(d: Record<string, string>) {
    const price = parseFloat(d["lastPrice"] ?? "0");
    const prev = parseFloat(d["prevPrice24h"] ?? d["lastPrice"] ?? "0");
    const bid = parseFloat(d["bid1Price"] ?? d["lastPrice"] ?? "0");
    const ask = parseFloat(d["ask1Price"] ?? d["lastPrice"] ?? "0");

    if (price > 0) {
      this._ticker = {
        exchange: "Bybit",
        symbol: SYMBOL,
        price,
        bid: bid || price,
        ask: ask || price,
        spread: ask > 0 && bid > 0 ? ask - bid : 0,
        priceChange24h: price - (prev || price),
        priceChangePct24h: parseFloat(d["price24hPcnt"] ?? "0") * 100,
        volume24h: parseFloat(d["volume24h"] ?? "0"),
        high24h: parseFloat(d["highPrice24h"] ?? "0"),
        low24h: parseFloat(d["lowPrice24h"] ?? "0"),
        openInterest: parseFloat(d["openInterest"] ?? "0"),
        fundingRate: parseFloat(d["fundingRate"] ?? "0"),
        nextFundingTime: parseInt(d["nextFundingTime"] ?? "0"),
        updatedAt: Date.now(),
      };
    }
  }

  private handleOrderbook(data: Record<string, unknown>, type: string) {
    if (type === "snapshot") { this.bidsMap.clear(); this.asksMap.clear(); }

    for (const [p, s] of (data["b"] as Array<[string, string]> ?? [])) {
      const sz = parseFloat(s);
      if (sz === 0) this.bidsMap.delete(p); else this.bidsMap.set(p, sz);
    }
    for (const [p, s] of (data["a"] as Array<[string, string]> ?? [])) {
      const sz = parseFloat(s);
      if (sz === 0) this.asksMap.delete(p); else this.asksMap.set(p, sz);
    }

    const bidsRaw = Array.from(this.bidsMap.entries())
      .map(([p, s]) => ({ price: parseFloat(p), size: s }))
      .sort((a, b) => b.price - a.price).slice(0, BOOK_LEVELS);
    const asksRaw = Array.from(this.asksMap.entries())
      .map(([p, s]) => ({ price: parseFloat(p), size: s }))
      .sort((a, b) => a.price - b.price).slice(0, BOOK_LEVELS);

    let bt = 0, at = 0;
    const bids: OrderbookLevel[] = bidsRaw.map(l => ({ ...l, total: (bt += l.size) }));
    const asks: OrderbookLevel[] = asksRaw.map(l => ({ ...l, total: (at += l.size) }));

    const spread = asks[0] && bids[0] ? asks[0].price - bids[0].price : 0;
    const mid = asks[0] && bids[0] ? (asks[0].price + bids[0].price) / 2 : 1;

    this._orderbook = {
      exchange: "Bybit",
      symbol: SYMBOL,
      bids,
      asks,
      spread,
      spreadPct: mid > 0 ? spread / mid * 100 : 0,
      imbalance: computeImbalance(bids, asks),
      updatedAt: Date.now(),
    };
  }

  private handleTrades(data: Array<Record<string, string>>) {
    for (const t of data) {
      this._trades.unshift({
        id: t["i"] ?? String(Date.now()),
        exchange: "Bybit",
        price: parseFloat(t["p"] ?? "0"),
        size: parseFloat(t["v"] ?? "0"),
        side: t["S"] === "Buy" ? "buy" : "sell",
        timestamp: parseInt(t["T"] ?? "0"),
      });
    }
    if (this._trades.length > MAX_TRADES) this._trades = this._trades.slice(0, MAX_TRADES);
  }

  private handleLiquidation(data: Record<string, unknown>) {

    };

    }
  }

function restRequest(
  method: "GET" | "POST" | "DELETE",
  path: string,
  params: Record<string, unknown> = {},
  apiKey: string,
  apiSecret: string,
): Promise<Record<string, unknown>> {
  return new Promise((resolve, reject) => {
    const timestamp = Date.now().toString();
    const paramStr = buildQueryString({ ...params, timestamp });
    const signature = hmacSha256(apiSecret, paramStr);
    const fullQuery = `${paramStr}&signature=${signature}`;
    const urlPath = method === "GET" || method === "DELETE" ? `${path}?${fullQuery}` : path;
    const body = method === "POST" ? fullQuery : "";

    const options = {
      hostname: "open-api.bingx.com",
      path: urlPath,
      method,
      headers: { "X-BX-APIKEY": apiKey, "Content-Type": "application/x-www-form-urlencoded" },
    };

    const req = https.request(options, (res) => {
      let data = "";
      res.on("data", (c) => (data += c));
      res.on("end", () => {
        try {
          const parsed = JSON.parse(data) as Record<string, unknown>;
          if (parsed["code"] !== 0 && parsed["code"] !== undefined) {
            reject(new Error(`BingX error ${parsed["code"]}: ${parsed["msg"]}`));
          } else {
            resolve((parsed["data"] ?? parsed) as Record<string, unknown>);
          }
        } catch (e) { reject(e); }
      });
    });
    req.on("error", reject);
    if (body) req.write(body);
    req.end();
  });
}

function sign(secret: string, ts: string, method: string, path: string, body = ""): string {
  return Buffer.from(
    hmacSha256(secret, `${ts}${method}${path}${body}`)
  ).toString("base64");
}

function restRequest(
  method: "GET" | "POST",
  path: string,
  params: Record<string, unknown> = {},
  apiKey: string,
  apiSecret: string,
  passphrase: string,
): Promise<unknown> {
  return new Promise((resolve, reject) => {
    const ts = new Date().toISOString();
    const queryString = method === "GET" && Object.keys(params).length > 0
      ? "?" + Object.entries(params).map(([k, v]) => `${k}=${v}`).join("&")
      : "";
    const bodyStr = method === "POST" ? JSON.stringify(params) : "";
    const fullPath = path + queryString;
    const sig = sign(apiSecret, ts, method, fullPath, bodyStr);

    const options = {
      hostname: "www.okx.com",
      path: fullPath,
      method,
      headers: {
        "OK-ACCESS-KEY": apiKey,
        "OK-ACCESS-SIGN": sig,
        "OK-ACCESS-TIMESTAMP": ts,
        "OK-ACCESS-PASSPHRASE": passphrase,
        "Content-Type": "application/json",
      },
    };

    const req = https.request(options, (res) => {
      let data = "";
      res.on("data", (c) => (data += c));
      res.on("end", () => {
        try {
          const parsed = JSON.parse(data) as Record<string, unknown>;
          if (parsed["code"] !== "0" && parsed["code"] !== undefined) {
            reject(new Error(`OKX error ${parsed["code"]}: ${parsed["msg"]}`));
          } else {
            resolve(parsed["data"] ?? parsed);
          }
        } catch (e) { reject(e); }
      });
    });

    req.on("error", reject);
    if (bodyStr) req.write(bodyStr);
    req.end();
  });
}

  async setLeverage(symbol: string, leverage: number): Promise<void> {
    if (!this.hasCredentials) return;
    await restRequest("POST", "/api/v5/account/set-leverage",
      { instId: INST_ID, lever: String(leverage), mgnMode: "cross" },
      this.apiKey, this.apiSecret, this.passphrase);
  }

  async placeOrder(params: PlaceOrderParams): Promise<OrderResult> {
    if (!this.hasCredentials) throw new Error("OKX credentials not configured");

    const body: Record<string, unknown> = {
      instId: INST_ID,
      tdMode: "cross",
      side: params.side === "Buy" ? "buy" : "sell",
      posSide: params.side === "Buy" ? "long" : "short",
      ordType: params.orderType === "Market" ? "market" : "limit",
      sz: String(params.qty),
    };
    if (params.price) body["px"] = String(params.price);
    if (params.stopLoss) body["slTriggerPx"] = String(params.stopLoss);
    if (params.takeProfit) body["tpTriggerPx"] = String(params.takeProfit);

    const res = await restRequest("POST", "/api/v5/trade/order", body, this.apiKey, this.apiSecret, this.passphrase) as Array<Record<string, unknown>>;
    const order = res?.[0] ?? {};
    logger.info({ order }, "OKX: order placed");

    return {
      orderId: String(order["ordId"] ?? Date.now()),
      exchange: "OKX",
      symbol: params.symbol,
      side: params.side,
      qty: params.qty,
      price: params.price ?? 0,
      status: String(order["sCode"] === "0" ? "created" : order["sMsg"]),
    };
  }

  async cancelOrder(orderId: string, _symbol: string): Promise<void> {
    if (!this.hasCredentials) return;
    await restRequest("POST", "/api/v5/trade/cancel-order",
      { instId: INST_ID, ordId: orderId }, this.apiKey, this.apiSecret, this.passphrase);
  }
}

import crypto from "crypto";

export function hmacSha256(secret: string, message: string): string {
  return crypto.createHmac("sha256", secret).update(message).digest("hex");
}

export function buildQueryString(params: Record<string, string | number | boolean>): string {
  return Object.entries(params)
    .filter(([, v]) => v !== undefined && v !== null && v !== "")
    .map(([k, v]) => `${encodeURIComponent(k)}=${encodeURIComponent(String(v))}`)
    .join("&");
}

export function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

export function computeImbalance(bids: { size: number }[], asks: { size: number }[], levels = 10): number {
  const bidVol = bids.slice(0, levels).reduce((s, l) => s + l.size, 0);
  const askVol = asks.slice(0, levels).reduce((s, l) => s + l.size, 0);
  const total = bidVol + askVol;
  return total > 0 ? (bidVol - askVol) / total : 0;
}

function MiniReturnChart({
  current,
  historical,
  forward,
}: {
  current: number[];
  historical: number[];
  forward: number[];
}) {
  const data = current.map((v, i) => ({
    day: i - current.length + 1,
    current: parseFloat((v * 100).toFixed(3)),
    historical: parseFloat((historical[i] * 100).toFixed(3)),
  }));
  const fwdData = forward.map((v, i) => ({
    day: i + 1,
    forward: parseFloat((v * 100).toFixed(3)),
  }));

  return (
    <div className="flex gap-1 h-12">
      <div className="flex-1">
        <ResponsiveContainer width="100%" height="100%">
          <LineChart data={data} margin={{ top: 2, right: 2, left: 0, bottom: 2 }}>
            <ReferenceLine y={0} stroke="rgba(255,255,255,0.15)" />
            <Line type="monotone" dataKey="current" stroke="#60a5fa" strokeWidth={1.5} dot={false} isAnimationActive={false} />
            <Line type="monotone" dataKey="historical" stroke="#a78bfa" strokeWidth={1} dot={false} strokeDasharray="3 2" isAnimationActive={false} />
            <Tooltip
              contentStyle={{ backgroundColor: '#1a1a2e', border: '1px solid #333', fontSize: '10px', padding: '2px 6px' }}
              formatter={(v: number) => [`${v}%`, ""]}
              labelFormatter={l => `День ${l}`}
            />
          </LineChart>
        </ResponsiveContainer>
      </div>
      <div className="w-16 border-l border-border/30">
        <ResponsiveContainer width="100%" height="100%">
          <LineChart data={fwdData} margin={{ top: 2, right: 2, left: 2, bottom: 2 }}>
            <ReferenceLine y={0} stroke="rgba(255,255,255,0.15)" />
            <Line
              type="monotone"
              dataKey="forward"
              stroke={fwdData.reduce((s, d) => s + d.forward, 0) >= 0 ? "#4ade80" : "#f87171"}
              strokeWidth={1.5}
              dot={false}
              isAnimationActive={false}
            />
          </LineChart>
        </ResponsiveContainer>
      </div>
    </div>
  );
}

  if (v >= 1_000_000) return `$${(v / 1_000_000).toFixed(1)}M`;
  if (v >= 1_000) return `$${(v / 1_000).toFixed(0)}K`;

}

function fmtBtc(v: number) {
  if (v >= 1) return `${v.toFixed(2)} BTC`;

}

              <LiqBar label="24H" longUsd={info.longVolUsd24h ?? 0} shortUsd={info.shortVolUsd24h ?? 0} />
            </CardContent>
          </Card>
        )}

        <div className="grid grid-cols-12 gap-2 flex-1 min-h-0">

          {/* LEFT: Liquidation Map */}
          <div className="col-span-5 flex flex-col min-h-0">
            <Card className="rounded-none border-border flex-1 flex flex-col min-h-0">
              <CardHeader className="p-3 pb-0 border-b border-border/50 shrink-0">
                <CardTitle className="text-xs uppercase flex justify-between items-start">
                  <div>
                    <div>Карта ликвидаций ±10% (Coinalyze)</div>
                    {currentPrice > 0 && (
                      <div className="text-[10px] font-normal text-muted-foreground mt-0.5 normal-case">
                        Текущая цена <span className="text-foreground font-mono">${currentPrice.toFixed(0)}</span>
                        {topClusters[0] && (
                          <span> · Топ кластер: <span className="font-mono">${topClusters[0].price.toFixed(0)}</span> ({fmt(topClusters[0].totalUsd)})</span>
                        )}
                      </div>
                    )}
                  </div>
                  <div className="flex flex-col gap-0.5 text-right shrink-0 ml-2">
                    <div className="flex items-center gap-1 justify-end"><span className="w-2 h-2 bg-green/70 inline-block" /> <span className="text-[9px] font-normal normal-case text-muted-foreground">Шорты (цена ↑)</span></div>
                    <div className="flex items-center gap-1 justify-end"><span className="w-2 h-2 bg-red/70 inline-block" /> <span className="text-[9px] font-normal normal-case text-muted-foreground">Лонги (цена ↓)</span></div>
                  </div>
                </CardTitle>
              </CardHeader>
              <CardContent className="p-0 flex-1 min-h-0">
                {mapChartData.length > 0 ? (
                  <ResponsiveContainer width="100%" height="100%">
                    <BarChart
                      data={mapChartData}
                      layout="vertical"
                      margin={{ top: 4, right: 8, left: 48, bottom: 4 }}
                      barCategoryGap={1}
                    >
                      <XAxis
                        type="number"
                        tick={{ fontSize: 8, fill: 'var(--color-muted-foreground)' }}
                        axisLine={false} tickLine={false}
                        tickFormatter={v => fmt(v)}
                      />
                      <YAxis
                        type="category" dataKey="priceLabel"
                        tick={{ fontSize: 8, fill: 'var(--color-muted-foreground)' }}
                        axisLine={false} tickLine={false}
                        width={46}
                      />
                      <Tooltip content={<MapTooltip />} />
                      <Bar dataKey="longLiqUsd" stackId="a" isAnimationActive={false} name="Лонги">
                        {mapChartData.map((entry, i) => (
                          <Cell
                            key={i}
                            fill={entry.isCurrentPrice ? '#ffffff' : 'var(--color-red)'}
                            fillOpacity={entry.isCurrentPrice ? 1 : 0.65}
                          />
                        ))}
                      </Bar>
                      <Bar dataKey="shortLiqUsd" stackId="a" isAnimationActive={false} name="Шорты">
                        {mapChartData.map((entry, i) => (
                          <Cell
                            key={i}
                            fill={entry.isCurrentPrice ? '#ffffff' : 'var(--color-green)'}
                            fillOpacity={entry.isCurrentPrice ? 0.3 : 0.65}
                          />
                        ))}
                      </Bar>
                    </BarChart>
                  </ResponsiveContainer>
                ) : (
                  <div className="h-full flex items-center justify-center text-muted-foreground text-sm">
                    Загрузка карты ликвидаций...
                  </div>
                )}
              </CardContent>
            </Card>
          </div>

          {/* CENTER: Historical chart + live feed header */}
          <div className="col-span-4 flex flex-col gap-2 min-h-0">
            <Card className="rounded-none border-border flex-1 flex flex-col min-h-0">
              <CardHeader className="p-3 pb-0 border-b border-border/50 shrink-0">
                <CardTitle className="text-xs uppercase flex justify-between items-center">
                  <span>История ликвидаций (все биржи)</span>
                  <div className="flex gap-1">
                    {(["24h", "2h"] as const).map(r => (
                      <button key={r} onClick={() => setHistRange(r)}
                        className={`px-2 py-0.5 text-[9px] border ${histRange === r ? "border-primary text-primary" : "border-border text-muted-foreground"}`}>
                        {r === "24h" ? "24H" : "2H"}
                      </button>
                    ))}
                  </div>
                </CardTitle>
              </CardHeader>
              <CardContent className="p-0 flex-1 min-h-0">
                {caChartData.length > 0 ? (
                  <ResponsiveContainer width="100%" height="100%">
                    <BarChart data={caChartData} margin={{ top: 8, right: 8, left: 4, bottom: 8 }}>
                      <XAxis dataKey="time" tick={{ fontSize: 7, fill: 'var(--color-muted-foreground)' }} axisLine={false} tickLine={false} interval="preserveStartEnd" />
                      <YAxis tick={{ fontSize: 8, fill: 'var(--color-muted-foreground)' }} axisLine={false} tickLine={false} tickFormatter={v => `${Math.abs(v).toFixed(1)}`} />
                      <Tooltip
                        contentStyle={{ backgroundColor: 'var(--color-card)', border: '1px solid var(--color-border)', fontSize: 10 }}
                        formatter={(v: number, name: string) => [`${Math.abs(v).toFixed(3)} BTC`, name === "longBtc" ? "🟢 Лонги ликвидированы" : "🔴 Шорты ликвидированы"]}
                      />
                      <Bar dataKey="longBtc" fill="var(--color-green)" fillOpacity={0.75} isAnimationActive={false} />
                      <Bar dataKey="shortBtc" fill="var(--color-red)" fillOpacity={0.75} isAnimationActive={false} />
                    </BarChart>
                  </ResponsiveContainer>
                ) : (
                  <div className="h-full flex items-center justify-center text-muted-foreground text-sm">Загрузка...</div>
                )}
              </CardContent>
            </Card>
          </div>

          {/* RIGHT: Stats + live feed */}
          <div className="col-span-3 flex flex-col gap-2 min-h-0">
            {info && (
              <Card className="rounded-none border-border shrink-0">
                <CardContent className="p-3 space-y-1">
                  <div className="text-[10px] uppercase text-muted-foreground mb-2">24h (CoinGlass)</div>
                  <div className="grid grid-cols-2 gap-2">
                    <div>
                      <div className="text-green font-bold font-mono text-sm">{fmt(info.longVolUsd24h ?? 0)}</div>
                      <div className="text-[10px] text-muted-foreground">Лонги ликв.</div>
                    </div>
                    <div>
                      <div className="text-red font-bold font-mono text-sm">{fmt(info.shortVolUsd24h ?? 0)}</div>
                      <div className="text-[10px] text-muted-foreground">Шорты ликв.</div>
                    </div>
                  </div>
                </CardContent>
              </Card>
            )}

            {/* Top clusters annotation */}
            {topClusters.length > 0 && (
              <Card className="rounded-none border-border shrink-0">
                <CardContent className="p-3">
                  <div className="text-[10px] uppercase text-muted-foreground mb-2">Топ кластеры</div>
                  {topClusters.map((b, i) => {
                    const isAbove = b.price > currentPrice;
                    const isBelow = b.price < currentPrice;
                    return (
                      <div key={i} className="flex justify-between items-center py-1 border-b border-border/20 last:border-0">
                        <div>
                          <div className="font-mono text-xs font-bold">${b.price.toFixed(0)}</div>
                          <div className={`text-[9px] ${isAbove ? 'text-green' : isBelow ? 'text-red' : 'text-muted-foreground'}`}>
                            {isAbove ? `↑ +${((b.price / currentPrice - 1) * 100).toFixed(1)}%` : isBelow ? `↓ -${((1 - b.price / currentPrice) * 100).toFixed(1)}%` : 'текущая'}
                          </div>
                        </div>
                        <div className="text-right">
                          <div className="text-xs font-bold font-mono">{fmt(b.totalUsd)}</div>
                          <div className="text-[9px] text-muted-foreground">
                            {b.longLiqUsd > b.shortLiqUsd ? <span className="text-red">Лонги</span> : <span className="text-green">Шорты</span>}
                          </div>
                        </div>
                      </div>
                    );
                  })}
                </CardContent>
              </Card>
            )}

            <Card className="rounded-none border-border flex-1 flex flex-col min-h-0">
              <CardHeader className="p-3 pb-0 border-b border-border/50 shrink-0">
                <CardTitle className="text-xs uppercase flex justify-between">
                  <span>Live (Bybit)</span>
                  <span className="text-muted-foreground font-normal">{eventList.length}</span>
                </CardTitle>
              </CardHeader>
              <CardContent className="p-0 flex-1 overflow-auto">
                <table className="w-full text-xs">
                  <thead className="sticky top-0 bg-card border-b border-border/50 text-muted-foreground">
                    <tr>
                      <th className="p-1.5 font-normal text-left">Время</th>
                      <th className="p-1.5 font-normal text-left">Side</th>
                      <th className="p-1.5 font-normal text-right pr-2">Сумма</th>
                    </tr>
                  </thead>
                  <tbody>
                    {!hasEvents ? (
                      <tr><td colSpan={3} className="p-6 text-center text-muted-foreground text-[10px]">Ждём ликвидаций...</td></tr>
                    ) : (
                      eventList.map((e, i) => (
                        <tr key={i} className={`border-b border-border/20 last:border-0 ${e.side === 'long' ? 'bg-green/5' : 'bg-red/5'}`}>
                          <td className="p-1.5 text-muted-foreground font-mono text-[10px]">{format(new Date(e.timestamp), 'HH:mm:ss')}</td>
                          <td className={`p-1.5 font-bold text-[10px] ${e.side === 'long' ? 'text-green' : 'text-red'}`}>{e.side.toUpperCase()}</td>
                          <td className={`p-1.5 pr-2 text-right font-bold font-mono text-[10px] ${e.side === 'long' ? 'text-green' : 'text-red'}`}>{fmt(e.valueUsd)}</td>
                        </tr>
                      ))
                    )}
                  </tbody>
                </table>
              </CardContent>
            </Card>
          </div>
        </div>
      </div>
    </Layout>
  );
}
